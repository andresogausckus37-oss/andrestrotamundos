function sincronizarEntorno(env) {
  const claves = [
    "MONGODB_URI",
    "OPENAI_API_KEY",
    "ADMIN_PASSWORD",
    "CRON_SECRET",
    "THREADS_APP_ID",
    "THREADS_APP_SECRET",
    "INSTAGRAM_ACCESS_TOKEN",
    "FACEBOOK_PAGE_ID",
    "FACEBOOK_PAGE_ACCESS_TOKEN",
    "TELEGRAM_VISITAS_BOT_TOKEN",
    "TELEGRAM_VISITAS_CHAT_ID",
    "RESEND_API_KEY",
  ];
  for (const clave of claves) {
    if (env?.[clave] !== undefined && env?.[clave] !== null) {
      process.env[clave] = String(env[clave]);
    }
  }
}

async function leerBody(request) {
  if (
    request.method === "GET" ||
    request.method === "HEAD"
  ) {
    return {};
  }

  const tipo =
    request.headers.get("content-type") || "";

  if (
    tipo.includes(
      "application/json"
    )
  ) {
    try {
      return await request.json();
    } catch {
      return {};
    }
  }

  if (
    tipo.includes(
      "multipart/form-data"
    )
  ) {
    try {
      const formData =
        await request.formData();

      return Object.fromEntries(
        formData.entries()
      );
    } catch {
      return {};
    }
  }

  if (
    tipo.includes(
      "application/x-www-form-urlencoded"
    )
  ) {
    try {
      return Object.fromEntries(
        new URLSearchParams(
          await request.text()
        )
      );
    } catch {
      return {};
    }
  }

  return {};
}

function headersComoObjeto(request) {
  const headers = {};
  for (const [k, v] of request.headers.entries()) headers[k.toLowerCase()] = v;
  headers.cookie = request.headers.get("cookie") || "";
  headers.authorization = request.headers.get("authorization") || "";
  return headers;
}

function crearRespuestaApi(origen) {
  let statusCode = 200;
  const headers = new Headers({ "Cache-Control": "no-store" });
  let final = null;
  return {
    status(c) { statusCode = Number(c) || 200; return this; },
    setHeader(n, v) { headers.set(n, String(v)); return this; },
    json(d) {
      headers.set("Content-Type", "application/json; charset=utf-8");
      final = new Response(JSON.stringify(d), { status: statusCode, headers });
      return final;
    },
    send(d) {
      if (d !== null && typeof d === "object" && !(d instanceof ArrayBuffer)) return this.json(d);
      final = new Response(d == null ? "" : String(d), { status: statusCode, headers });
      return final;
    },
    redirect(codigoOUrl, urlOpcional) {
      let codigo = 302, url = codigoOUrl;
      if (typeof codigoOUrl === "number") { codigo = codigoOUrl; url = urlOpcional; }
      final = Response.redirect(new URL(String(url), origen).toString(), codigo);
      return final;
    },
    end() { final = new Response(null, { status: statusCode, headers }); return final; },
    obtenerRespuesta() { return final; }
  };
}

async function ejecutarHandlerContenidoRedes(request, env) {
  sincronizarEntorno(env);
  const url = new URL(request.url);
  const req = {
    method: request.method,
    url: request.url,
    query: Object.fromEntries(url.searchParams.entries()),
    body: await leerBody(request),
    headers: headersComoObjeto(request),
    env: {
      IMAGES: env.IMAGES,
      PRODUCTOS_R2: env.PRODUCTOS_R2,
      PRODUCTOS_R2_URL: "https://pub-60d59fb304eb48d384739f40d6f48d4f.r2.dev",
    },
  };
  const res = crearRespuestaApi(url.origin);
  const retorno = await handlerContenidoRedes(req, res);
  if (retorno instanceof Response) return retorno;
  return res.obtenerRespuesta() || Response.json({ ok:false, error:"La API no generó una respuesta." }, { status:500 });
}

async function ejecutarHandlerGenerico(request, env, handler) {
  sincronizarEntorno(env);

  const url = new URL(request.url);

  const req = {
    method: request.method,
    url: request.url,
    query: Object.fromEntries(
      url.searchParams.entries()
    ),
    body: await leerBody(request),
    headers: headersComoObjeto(request),

    env: {
      PRODUCTOS_R2: env.PRODUCTOS_R2,
      PRODUCTOS_R2_URL:
        "https://pub-60d59fb304eb48d384739f40d6f48d4f.r2.dev",
    },
  };

  const res = crearRespuestaApi(url.origin);

  const retorno = await handler(req, res);

  if (retorno instanceof Response) {
    return retorno;
  }

  return (
    res.obtenerRespuesta() ||
    Response.json(
      { ok: false, error: "La API no generó una respuesta." },
      { status: 500 }
    )
  );
}

function respuestaServicio(datos, estado = 200) {
  return Response.json(datos, {
    status: estado,
    headers: { "Cache-Control": "no-store" },
  });
}

async function leerJsonServicio(request) {
  if (!/^application\/json(?:;|$)/i.test(request.headers.get("content-type") || "")) {
    return { error: "El pedido debe enviarse como JSON.", estado: 415 };
  }
  const lector = request.body?.getReader();
  if (!lector) return { error: "Falta la configuración del pedido.", estado: 400 };
  const decoder = new TextDecoder();
  let texto = "", bytes = 0;
  try {
    while (true) {
      const { done, value } = await lector.read();
      if (done) break;
      bytes += value.byteLength;
      if (bytes > 64 * 1024) {
        await lector.cancel();
        return { error: "El pedido supera el tamaño permitido.", estado: 413 };
      }
      texto += decoder.decode(value, { stream: true });
    }
    texto += decoder.decode();
    const datos = JSON.parse(texto);
    if (!datos || typeof datos !== "object" || Array.isArray(datos)) {
      return { error: "Formato del pedido inválido.", estado: 400 };
    }
    return { datos };
  } catch {
    return { error: "No se pudo leer el JSON del pedido.", estado: 400 };
  }
}

function textoServicio(valor, maximo) {
  return typeof valor === "string" && valor.trim().length <= maximo
    ? valor.trim()
    : null;
}

function urlServicio(valor, maximo, soloHttps = false) {
  const texto = textoServicio(valor, maximo);
  if (texto === null) return null;
  if (!texto) return "";
  try {
    const url = new URL(texto);
    if ((soloHttps ? url.protocol !== "https:" : !["http:", "https:"].includes(url.protocol)) ||
        url.username || url.password || !url.hostname) return null;
    return url.toString();
  } catch {
    return null;
  }
}

// Acepta enlaces https://, www.dominio.com o dominio.com sin reescribirlos.
// Valida una URL segura para evitar esquemas peligrosos al mostrarla después.
function enlaceEscritoServicio(valor, maximo) {
  const texto = textoServicio(valor, maximo);
  if (texto === null || /[\x00-\x1f\x7f]/.test(texto)) return null;
  if (!texto) return "";
  try {
    const tieneProtocolo = /^[a-z][a-z0-9+.-]*:/i.test(texto);
    const destino = new URL(tieneProtocolo ? texto : `https://${texto}`);
    if (!["http:", "https:"].includes(destino.protocol) || destino.username || destino.password || !destino.hostname) return null;
    return texto;
  } catch { return null; }
}

// Los datos recibidos se copian campo por campo: no se persisten claves extra.
function validarConfiguracionServicio(datos) {
  if (datos.planId !== PLAN_PUBLICACIONES.id) {
    return { error: "El plan solicitado no existe." };
  }
  const redes = datos.redes;
  if (!Array.isArray(redes) || redes.length < 1 || redes.length > 2 ||
      new Set(redes).size !== redes.length ||
      !redes.every((red) => red === "instagram" || red === "facebook")) {
    return { error: "Seleccioná Instagram, Facebook o ambas redes." };
  }
  if (!ZONAS_PUBLICACIONES.has(datos.zonaHoraria)) {
    return { error: "Zona horaria no admitida." };
  }

  const n = datos.negocio;
  if (!n || typeof n !== "object" || Array.isArray(n)) {
    return { error: "Faltan los datos del emprendimiento." };
  }
  const negocio = {
    nombre: textoServicio(n.nombre, 120),
    rubro: textoServicio(n.rubro ?? "", 100),
    contacto: textoServicio(n.contacto, 120),
    email: textoServicio(n.email, 160),
    url: enlaceEscritoServicio(n.url ?? "", 300),
    whatsapp: textoServicio(n.whatsapp ?? "", 50),
    publico: textoServicio(n.publico ?? "", 180),
    tono: n.tono ?? "cercano",
  };
  if (!negocio.nombre || !negocio.contacto || !negocio.email ||
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(negocio.email) ||
      Object.values(negocio).some((valor) => valor === null) ||
      !["cercano", "profesional", "informal", "promocional"].includes(negocio.tono)) {
    return { error: "Revisá los datos del emprendimiento y el correo electrónico." };
  }
  if (negocio.whatsapp && !/^[+\d()\s.-]{5,50}$/.test(negocio.whatsapp)) {
    return { error: "El número de WhatsApp no es válido." };
  }

  if (!Array.isArray(datos.publicaciones) || datos.publicaciones.length !== PLAN_PUBLICACIONES.contenidos) {
    return { error: "El plan incluye exactamente cuatro contenidos." };
  }
  const publicaciones = datos.publicaciones.map((p, indice) => {
    if (!p || typeof p !== "object" || Array.isArray(p) ||
        p.id !== `contenido-${indice + 1}` ||
        !["reel", "carrusel", "imagen"].includes(p.tipo) ||
        !DIAS_PUBLICACIONES.has(p.dia) || !HORAS_PUBLICACIONES.has(p.hora)) return null;
    if (!(p.tipo === "reel" ? ["9:16"] : ["4:5", "1:1"]).includes(p.formato)) return null;

    const descripcion = textoServicio(p.descripcion ?? "", 1500);
    const ctaInstagram = textoServicio(p.ctaInstagram ?? "", 100);
    const ctaFacebook = textoServicio(p.ctaFacebook ?? "", 100);
    const urlFacebook = enlaceEscritoServicio(p.urlFacebook ?? "", 500);
    const hashtags = textoServicio(p.hashtags ?? "", 240);
    if ([descripcion, ctaInstagram, ctaFacebook, urlFacebook, hashtags].some((v) => v === null)) return null;

    const archivos = p.archivosSeleccionados ?? [];
    if (!Array.isArray(archivos) || archivos.length > (p.tipo === "carrusel" ? 10 : 1)) return null;
    const metadatos = archivos.map((archivo) => {
      if (!archivo || typeof archivo !== "object" || Array.isArray(archivo)) return null;
      const nombre = textoServicio(archivo.nombre, 180);
      const tipo = textoServicio(archivo.tipo, 80);
      const limite = p.tipo === "reel" ? 150 * 1024 * 1024 : 20 * 1024 * 1024;
      if (!nombre || tipo === null || !Number.isSafeInteger(archivo.tamano) ||
          archivo.tamano <= 0 || archivo.tamano > limite) return null;
      if (p.tipo === "reel" && tipo !== "video/mp4" && !/\.mp4$/i.test(nombre)) return null;
      if (p.tipo !== "reel" && !["image/jpeg", "image/png", "image/webp", ""].includes(tipo) &&
          !/\.(jpe?g|png|webp)$/i.test(nombre)) return null;
      return { nombre, tipo, tamano: archivo.tamano, estado: "pendiente_subida" };
    });
    if (metadatos.some((archivo) => !archivo)) return null;
    return {
      id: p.id, tipo: p.tipo, formato: p.formato,
      diaPreferido: p.dia, horaLocalPreferida: p.hora,
      descripcion, ctaInstagram, ctaFacebook, urlFacebook, hashtags,
      archivos: metadatos,
      fechaHoraDefinitiva: null,
      estado: "pendiente_preparacion",
    };
  });
  if (publicaciones.some((publicacion) => !publicacion)) {
    return { error: "Revisá los datos de las cuatro publicaciones." };
  }
  return { configuracion: {
    redes: [...redes], zonaHoraria: datos.zonaHoraria, negocio, publicaciones,
  } };
}

// Para abrir pedidos públicos se requiere Turnstile; Admin puede probar
// la escritura con su sesión existente. Esta verificación NO confirma pagos.
// Limitadores nativos de Cloudflare Workers. Protegen las rutas de servicios;
// los pedidos de la tienda y el calendario automatico no utilizan estos bindings.
function limitadoresServicioPreparados(env) {
  return [
    env?.SERVICIO_LIMITE_PEDIDOS,
    env?.SERVICIO_LIMITE_CONSULTAS,
    env?.SERVICIO_LIMITE_COMPROBANTES,
  ].every((binding) => typeof binding?.limit === "function");
}

async function comprobarLimiteServicio(request, env, nombreBinding, grupoAcotado = "") {
  const limitador = env?.[nombreBinding];
  if (typeof limitador?.limit !== "function") {
    return respuestaServicio({ ok: false, error: "La proteccion de solicitudes aun no esta configurada." }, 503);
  }
  // Creacion publica sin sesion: usar IP de Cloudflare (no X-Forwarded-For).
  // El limite es aproximado; usuarios que comparten IP pueden verse afectados.
  const ip = request.headers.get("CF-Connecting-IP")?.trim();
  if (!ip) {
    return respuestaServicio({ ok: false, error: "No se pudo verificar el origen de la solicitud." }, 403);
  }
  const clave = crypto.createHash("sha256").update(grupoAcotado ? `${ip}:${grupoAcotado}` : ip).digest("hex");
  try {
    const resultado = await limitador.limit({ key: clave });
    if (!resultado?.success) {
      return respuestaServicio({
        ok: false,
        error: "Se realizaron demasiadas solicitudes. Espera un minuto e intenta nuevamente.",
      }, 429);
    }
    return null;
  } catch (error) {
    console.error("Error del limitador de servicios:", error);
    return respuestaServicio({ ok: false, error: "La proteccion de solicitudes no esta disponible." }, 503);
  }
}

async function validarAccesoCrearServicio(request, env, datos) {
  if (adminAutorizadoWorker(request, env)) return { ok: true };
  if (!env.SERVICIO_TURNSTILE_SECRET) {
    return { error: "Aún no está habilitada la validación de pedidos públicos.", estado: 503 };
  }
  const token = textoServicio(datos.turnstileToken, 2048);
  if (!token) return { error: "Completá la verificación de seguridad.", estado: 403 };
  try {
    const formulario = new URLSearchParams({
      secret: env.SERVICIO_TURNSTILE_SECRET,
      response: token,
    });
    const ip = request.headers.get("CF-Connecting-IP");
    if (ip) formulario.set("remoteip", ip);
    const respuesta = await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
      method: "POST", body: formulario,
    });
    if (!respuesta.ok) return { error: "No se pudo completar la verificación de seguridad.", estado: 503 };
    const verificacion = await respuesta.json();
    // El token debe haberse emitido para el mismo dominio de este sitio.
    const hostnameEsperado = new URL(request.url).hostname.toLowerCase();
    const hostnameVerificado = String(verificacion.hostname ?? "").toLowerCase();
    return verificacion.success && hostnameVerificado === hostnameEsperado
      ? { ok: true }
      : { error: "Verificación de seguridad inválida, vencida o de otro dominio.", estado: 403 };
  } catch {
    return { error: "No se pudo comprobar la verificación de seguridad.", estado: 503 };
  }
}

// El cliente ya pasó Turnstile en el registro y verificó el correo por OTP.
// Un flag independiente sigue protegiendo la habilitación de las pruebas.
function preparacionPruebaDisponible(request, env, protegida = false) {
  if (String(env?.SERVICIO_PRUEBAS_PREPARACION_HABILITADA ?? "").toLowerCase() !== "true") return false;
  if (protegida && !limitadoresServicioPreparados(env)) return false;
  return true;
}

async function apiPruebaPublicaciones(request, env, ruta) {
  const esEstado = ruta === "/api/servicios/publicaciones/prueba/estado";
  if (request.method !== (esEstado ? "GET" : "POST")) {
    return respuestaServicio({ ok: false, error: "Método no permitido." }, 405);
  }
  if (esEstado) {
    const pedidoId = new URL(request.url).searchParams.get("pedidoId");
    if (!pedidoServicioIdValido(pedidoId)) return respuestaServicio({ ok: false, error: "Pedido inválido." }, 400);
    const exceso = await comprobarLimiteServicio(request, env, "SERVICIO_LIMITE_CONSULTAS");
    if (exceso) return exceso;
    try {
      return await conColeccionServicios(env, async coleccion => {
        const pedido = await coleccion.findOne({ pedidoId, tipo: "publicaciones", modalidadContratacion: "prueba_gratuita" });
        if (!pedido || !clienteServicioAutorizado(request, pedido)) return respuestaServicio({ ok: false, error: "Pedido no encontrado o acceso no autorizado." }, 404);
        return respuestaServicio({ ok: true, pedidoId, modalidad: "prueba_gratuita", moneda: pedido.moneda,
          precio: 0, precioRenovacion: pedido.precioRenovacion, metodoPago: "ninguno",
          estado: pedido.estado, estadoServicio: pedido.estadoServicio, estadoContenidos: pedido.estadoContenidos,
          prueba: { estado: pedido.prueba?.estado || "pendiente_activacion", inicio: pedido.prueba?.inicio || null,
            fin: pedido.prueba?.fin || null }, pagoHabilitado: false });
      });
    } catch {
      return respuestaServicio({ ok: false, error: "No se pudo consultar la prueba." }, 503);
    }
  }

  if (!preparacionPruebaDisponible(request, env, true)) {
    return respuestaServicio({ ok: false, error: "La preparación de pruebas todavía no está habilitada." }, 503);
  }
  // Protección por IP adicional a la sesión verificada del titular.
  const exceso = await comprobarLimiteServicio(request, env, "SERVICIO_LIMITE_PEDIDOS");
  if (exceso) return exceso;
  if (request.headers.get("origin") !== new URL(request.url).origin) {
    return respuestaServicio({ ok: false, error: "Origen no autorizado." }, 403);
  }
  const lectura = await leerJsonServicio(request);
  if (lectura.error) return respuestaServicio({ ok: false, error: lectura.error }, lectura.estado);
  const datos = lectura.datos;
  if (!["ARS", "USD"].includes(datos.moneda)) {
    return respuestaServicio({ ok: false, error: "Moneda no admitida." }, 400);
  }
  // La segunda verificación de Cloudflare no es necesaria: la prueba solo
  // puede crearla una cuenta con correo validado y sesión activa.
  let usuario = null;
  try { usuario = await obtenerClienteRegistrado(request, env); }
  catch { return respuestaServicio({ ok: false, error: "No se pudo verificar la sesión de cliente." }, 503); }
  if (!usuario?.id) return respuestaServicio({ ok: false, error: "Iniciá sesión con tu correo confirmado." }, 401);
  const validacion = validarConfiguracionServicio(datos);
  if (validacion.error) return respuestaServicio({ ok: false, error: validacion.error }, 400);
  if (usuario.email !== validacion.configuracion.negocio.email.toLowerCase()) {
    return respuestaServicio({ ok: false, error: "Usá el correo verificado de tu cuenta para iniciar la prueba." }, 400);
  }

  let mongo, reservado = false;
  const pedidoId = `SER-${crypto.randomUUID()}`;
  const fecha = new Date();
  // Una sola prueba por cuenta con correo verificado.

// =========================================================
// TRANSFERENCIAS DE PUBLICACIONES · ETAPA 2
// Independientes de "pedidos" (tienda). NO procesa cobros automáticos.
// Comprobantes únicamente en un bucket R2 PRIVADO.
// =========================================================
const ESTADOS_TRANSFERENCIA_SERVICIO = Object.freeze({
  ESPERANDO: "pendiente_comprobante",
  COMPROBANTE: "verificando_comprobante",
  VERIFICANDO: "verificando_pago",
  COMPLETANDO: "completando_operacion",
});

function transferenciasServiciosHabilitadas(env) {
  return String(env?.SERVICIO_TRANSFERENCIAS_HABILITADAS ?? "").toLowerCase() === "true";
}

function cuentaTransferenciaServicio(env) {
  const titular = String(env?.SERVICIO_TRANSFERENCIA_TITULAR ?? "").trim();
  const entidad = String(env?.SERVICIO_TRANSFERENCIA_ENTIDAD ?? "").trim();
  const alias = String(env?.SERVICIO_TRANSFERENCIA_ALIAS ?? "").trim();
  const cbuCvu = String(env?.SERVICIO_TRANSFERENCIA_CBU_CVU ?? "").replace(/\s/g, "");
  if (!titular || titular.length > 120 || !entidad || entidad.length > 80 ||
      !/^[\w.-]{6,60}$/.test(alias) || !/^\d{22}$/.test(cbuCvu)) return null;
  return { titular, entidad, alias, cbuCvu };
}

function pedidoServicioIdValido(valor) {
  return typeof valor === "string" && /^SER-[a-f0-9-]{36}$/i.test(valor);
}

function tokenClienteServicio(request) {
  const autorizacion = request.headers.get("authorization") || "";
  const match = /^Bearer ([a-f0-9]{64})$/i.exec(autorizacion);
  return match?.[1] || null;
}

function clienteServicioAutorizado(request, pedido) {
  const token = tokenClienteServicio(request);
  const hashGuardado = pedido?.tokenAccesoHash;
  if (!token || typeof hashGuardado !== "string" || !/^[a-f0-9]{64}$/i.test(hashGuardado)) return false;
  const calculado = crypto.createHash("sha256").update(token).digest("hex");
  return crypto.timingSafeEqual(Buffer.from(calculado, "hex"), Buffer.from(hashGuardado, "hex"));
}

async function conColeccionServicios(env, accion) {
  sincronizarEntorno(env);
  if (!env?.MONGODB_URI) throw new Error("MongoDB no configurado.");
  const { MongoClient } = await import("mongodb");
  const cliente = new MongoClient(env.MONGODB_URI);
  try {
    await cliente.connect();
    return await accion(cliente.db("andres_imprimibles").collection("pedidos_servicios"));
  } finally {
    await cliente.close().catch(() => {});
  }
}

function bucketComprobantesServicio(env) {
  // Nunca usar PRODUCTOS_R2: su contenido puede estar expuesto públicamente.
  const bucket = env?.SERVICIOS_COMPROBANTES_R2;
  return bucket?.put && bucket?.get ? bucket : null;
}

async function leerArchivoComprobanteServicio(request, maxBytes) {
  const declarado = Number(request.headers.get("content-length") || 0);
  if (declarado > maxBytes) return null;
  if (!request.body) return null;
  const lector = request.body.getReader();
  let total = 0;
  const partes = [];
  while (true) {
    const { done, value } = await lector.read();
    if (done) break;
    total += value.byteLength;
    if (total > maxBytes) { await lector.cancel(); return null; }
    partes.push(value);
  }
  if (total < 100) return null;
  const resultado = new Uint8Array(total);
  let posicion = 0;
  for (const parte of partes) { resultado.set(parte, posicion); posicion += parte.byteLength; }
  return resultado;
}

function tipoComprobanteServicio(bytes) {
  if (bytes[0] === 0xff && bytes[1] === 0xd8 && bytes[2] === 0xff) {
    return { tipo: "image/jpeg", extension: "jpg" };
  }
  if ([137, 80, 78, 71, 13, 10, 26, 10].every((b, i) => bytes[i] === b)) {
    return { tipo: "image/png", extension: "png" };
  }
  if (bytes[0] === 37 && bytes[1] === 80 && bytes[2] === 68 && bytes[3] === 70 && bytes[4] === 45) {
    return { tipo: "application/pdf", extension: "pdf" };
  }
  return null;
}

async function apiTransferenciaServicio(request, env, ruta) {
  if (!transferenciasServiciosHabilitadas(env)) {
    return respuestaServicio({ ok: false, error: "Las transferencias de publicaciones aún no están habilitadas." }, 503);
  }
  const cuenta = cuentaTransferenciaServicio(env);
  if (!cuenta) return respuestaServicio({ ok: false, error: "La cuenta bancaria todavía no está configurada." }, 503);

  const subir = ruta.endsWith("/comprobante");
  if (request.method !== (subir ? "POST" : "GET")) {
    return respuestaServicio({ ok: false, error: "Método no permitido." }, 405);
  }
  const pedidoId = new URL(request.url).searchParams.get("pedidoId");
  if (!pedidoServicioIdValido(pedidoId)) return respuestaServicio({ ok: false, error: "Identificador de pedido inválido." }, 400);
  const exceso = await comprobarLimiteServicio(request, env,
    subir ? "SERVICIO_LIMITE_COMPROBANTES" : "SERVICIO_LIMITE_CONSULTAS");
  if (exceso) return exceso;
  try {
    return await conColeccionServicios(env, async (coleccion) => {
      const pedido = await coleccion.findOne({ pedidoId, tipo: "publicaciones", metodoPago: "transferencia", moneda: "ARS" });
      if (!pedido || !clienteServicioAutorizado(request, pedido)) {
        return respuestaServicio({ ok: false, error: "Pedido no encontrado o acceso no autorizado." }, 404);
      }
      if (!subir) {
        return respuestaServicio({
          ok: true, pedidoId, moneda: "ARS", precio: pedido.precio,
          estado: pedido.estado, cuenta,
          comprobanteRecibido: Boolean(pedido.comprobante),
        });
      }
      if (pedido.estado !== ESTADOS_TRANSFERENCIA_SERVICIO.ESPERANDO) {
        return respuestaServicio({ ok: false, error: "Este pedido ya tiene un comprobante o no admite nuevas cargas." }, 409);
      }
      const bucket = bucketComprobantesServicio(env);
      if (!bucket) return respuestaServicio({ ok: false, error: "El almacenamiento privado de comprobantes no está preparado." }, 503);
      if (!/^(image\/jpeg|image\/png|application\/pdf)(?:;|$)/i.test(request.headers.get("content-type") || "")) {
        return respuestaServicio({ ok: false, error: "Adjuntá una imagen JPG/PNG o un PDF." }, 415);
      }
      const bytes = await leerArchivoComprobanteServicio(request, 8 * 1024 * 1024);
      if (!bytes) return respuestaServicio({ ok: false, error: "El comprobante debe ocupar entre 100 bytes y 8 MB." }, 413);
      const formato = tipoComprobanteServicio(bytes);
      if (!formato) return respuestaServicio({ ok: false, error: "El comprobante no es un JPG, PNG o PDF válido." }, 415);
      const fecha = new Date();
      const clave = `comprobantes/publicaciones/${pedidoId}/${crypto.randomUUID()}.${formato.extension}`;
      await bucket.put(clave, bytes, { httpMetadata: { contentType: formato.tipo } });
      try {
        const resultado = await coleccion.updateOne(
          { _id: pedido._id, estado: ESTADOS_TRANSFERENCIA_SERVICIO.ESPERANDO, comprobante: null },
          {
            $set: {
              comprobante: { r2Key: clave, tipo: formato.tipo, tamano: bytes.byteLength, subidoEn: fecha },
              estado: ESTADOS_TRANSFERENCIA_SERVICIO.COMPROBANTE,
              estadoPago: "verificando_comprobante", actualizadoEn: fecha,
            },
            $push: { historialEstados: { estado: ESTADOS_TRANSFERENCIA_SERVICIO.COMPROBANTE, fecha } },
          }
        );
        if (resultado.modifiedCount !== 1) {
          await bucket.delete(clave);
          return respuestaServicio({ ok: false, error: "El pedido cambió de estado. Actualizá la página." }, 409);
        }
      } catch (error) { await bucket.delete(clave).catch(() => {}); throw error; }
      return respuestaServicio({ ok: true, pedidoId, estado: ESTADOS_TRANSFERENCIA_SERVICIO.COMPROBANTE }, 201);
    });
  } catch (error) {
    console.error("Error en transferencia de publicaciones:", error);
    return respuestaServicio({ ok: false, error: "No se pudo procesar la transferencia del servicio." }, 500);
  }
}

async function apiAdminTransferenciasServicio(request, env) {
  if (!adminAutorizadoWorker(request, env)) return respuestaServicio({ ok: false, error: "No autorizado." }, 401);
  const url = new URL(request.url);
  const accion = url.searchParams.get("accion") || "listar";
  if (!((accion === "listar" || accion === "comprobante") && request.method === "GET") &&
      !(accion === "avanzar" && request.method === "POST")) {
    return respuestaServicio({ ok: false, error: "Método o acción no permitidos." }, 405);
  }
  // La sesión admin usa cookies; para mutaciones, exigir mismo origen como defensa CSRF.
  if (accion === "avanzar" && request.headers.get("origin") !== url.origin) {
    return respuestaServicio({ ok: false, error: "Origen no autorizado." }, 403);
  }
  try {
    return await conColeccionServicios(env, async (coleccion) => {
      if (accion === "listar") {
        const pedidos = await coleccion.find({ tipo: "publicaciones", metodoPago: "transferencia", pedidoId: /^SER-/ })
          .sort({ creadoEn: -1 }).limit(100)
          .project({ _id: 0, pedidoId: 1, precio: 1, moneda: 1, estado: 1, creadoEn: 1,
            "negocio.nombre": 1, "negocio.email": 1, comprobante: 1 }).toArray();
        return respuestaServicio({ ok: true, pedidos: pedidos.map((p) => ({
          pedidoId: p.pedidoId, precio: p.precio, moneda: p.moneda, estado: p.estado,
          creadoEn: p.creadoEn, negocio: p.negocio,
          comprobanteRecibido: Boolean(p.comprobante),
        })) });
      }
      let pedidoId;
      if (accion === "comprobante") pedidoId = url.searchParams.get("pedidoId");
      else {
        const lectura = await leerJsonServicio(request);
        if (lectura.error) return respuestaServicio({ ok: false, error: lectura.error }, lectura.estado);
        pedidoId = lectura.datos.pedidoId;
      }
      if (!pedidoServicioIdValido(pedidoId)) return respuestaServicio({ ok: false, error: "Pedido inválido." }, 400);
      const pedido = await coleccion.findOne({ pedidoId, tipo: "publicaciones", metodoPago: "transferencia" });
      if (!pedido) return respuestaServicio({ ok: false, error: "Pedido no encontrado." }, 404);
      if (accion === "comprobante") {
        if (!pedido.comprobante?.r2Key) return respuestaServicio({ ok: false, error: "El pedido no tiene comprobante." }, 404);
        const bucket = bucketComprobantesServicio(env);
        if (!bucket) return respuestaServicio({ ok: false, error: "Bucket privado no disponible." }, 503);
        const objeto = await bucket.get(pedido.comprobante.r2Key);
        if (!objeto) return respuestaServicio({ ok: false, error: "No se encontró el comprobante." }, 404);
        return new Response(objeto.body, { headers: {
          "Content-Type": pedido.comprobante.tipo,
          "Content-Disposition": `inline; filename="comprobante-${pedidoId}.${pedido.comprobante.tipo === "application/pdf" ? "pdf" : pedido.comprobante.tipo === "image/png" ? "png" : "jpg"}"`,
          "Cache-Control": "private, no-store", "X-Content-Type-Options": "nosniff",
        } });
      }
      let proximo, estadoPago;
      if (pedido.estado === ESTADOS_TRANSFERENCIA_SERVICIO.COMPROBANTE) {
        proximo = ESTADOS_TRANSFERENCIA_SERVICIO.VERIFICANDO;
        estadoPago = "verificando";
      } else if (pedido.estado === ESTADOS_TRANSFERENCIA_SERVICIO.VERIFICANDO) {
        proximo = ESTADOS_TRANSFERENCIA_SERVICIO.COMPLETANDO;
        estadoPago = "confirmado";
      } else {
        return respuestaServicio({ ok: false, error: "Este pedido no admite avanzar de estado." }, 409);
      }
      const fecha = new Date();
      const cambios = { estado: proximo, estadoPago, actualizadoEn: fecha };
      if (proximo === ESTADOS_TRANSFERENCIA_SERVICIO.COMPLETANDO) cambios.pagadoEn = fecha;
      const resultado = await coleccion.updateOne(
        { _id: pedido._id, estado: pedido.estado, "comprobante.r2Key": { $exists: true } },
        { $set: cambios, $push: { historialEstados: { estado: proximo, fecha } } }
      );
      if (resultado.modifiedCount !== 1) {
        return respuestaServicio({ ok: false, error: "El estado cambió. Actualizá el panel." }, 409);
      }
      // Pago confirmado != servicio activado. El contenido requiere aprobación posterior.
      return respuestaServicio({ ok: true, pedidoId, estado: proximo, estadoPago, servicioActivado: false });
    });
  } catch (error) {
    console.error("Error en administración de transferencias de servicios:", error);
    return respuestaServicio({ ok: false, error: "No se pudo completar la operación." }, 500);
  }
}

  if (original.startsWith(prefijo) && env.PRODUCTOS_R2?.get) {
    const clave = decodeURIComponent(new URL(original).pathname.slice(1));
    if (!clave || clave.includes("..")) return respuestaStoryError("Clave R2 no válida", 400);
    const objeto = await env.PRODUCTOS_R2.get(clave);
    if (!objeto) return respuestaStoryError("La imagen original no existe en R2", 404);
    tipo = objeto.httpMetadata?.contentType || "image/jpeg";
    bytes = await objeto.arrayBuffer();
  } else {
    const url = new URL(original);
    if (url.protocol !== "https:" || url.username || url.password) {
      return respuestaStoryError("URL de imagen no válida", 400);
    }
    const respuesta = await fetch(url.toString(), { redirect: "error" });
    if (!respuesta.ok) return respuestaStoryError(`Error descargando la imagen original (${respuesta.status})`, 502);
    tipo = respuesta.headers.get("content-type") || "";
    if (!/^image\/(png|jpeg|webp)(?:;|$)/i.test(tipo)) {
      return respuestaStoryError("El recurso original no es JPG, PNG o WebP", 422);
    }
    bytes = await respuesta.arrayBuffer();
  }
  if (bytes.byteLength < 100 || bytes.byteLength > 12 * 1024 * 1024) {
    return respuestaStoryError("La imagen original debe pesar menos de 12 MB", 413);
  }
  if (!/^image\/(png|jpeg|webp)/i.test(tipo)) {
    return respuestaStoryError("Formato original no admitido", 422);
  }
  return new Response(bytes, {
    status: 200,
    headers: {
      "Content-Type": tipo,
      "Cache-Control": "private, no-store",
      "X-Content-Type-Options": "nosniff",
    },
  });
}

async function subirImagenStoryCanvas(request, env) {
  if (!adminAutorizadoWorker(request, env)) {
    return respuestaStoryError("No autorizado", 401);
  }
  if (request.method !== "POST") return respuestaStoryError("Método no permitido", 405);
  const id = String(new URL(request.url).searchParams.get("productoId") || "").trim();
  if (!id || id.length > 160 || !/^[a-zA-Z0-9_-]+$/.test(id)) {
    return respuestaStoryError("ID de producto no válido", 400);
  }
  if (!env.PRODUCTOS_R2?.put || !env.PRODUCTOS_R2?.head) {
    return respuestaStoryError("Falta el binding PRODUCTOS_R2", 503);
  }
  const producto = await obtenerProductoStory(id, env);
  if (!producto) return respuestaStoryError("El producto no existe", 404);
  if (!String(producto.imagenes?.redes?.vertical?.presentacion || "").startsWith("https://")) {
    return respuestaStoryError("El producto no tiene una imagen vertical pública", 422);
  }
  if (!/^image\/jpeg(?:;|$)/i.test(request.headers.get("content-type") || "")) {
    return respuestaStoryError("Se requiere un JPEG generado por Canvas", 415);
  }
  const longitud = Number(request.headers.get("content-length") || 0);
  if (longitud > 8 * 1024 * 1024) return respuestaStoryError("JPEG demasiado grande", 413);
  const bytes = await request.arrayBuffer();
  if (bytes.byteLength < 100 || bytes.byteLength > 8 * 1024 * 1024) {
    return respuestaStoryError("JPEG vacío o demasiado grande", 413);
  }
  const firma = new Uint8Array(bytes);
  if (firma[0] !== 0xff || firma[1] !== 0xd8 || firma[2] !== 0xff) {
    return respuestaStoryError("El archivo no es un JPEG válido", 415);
  }
  const hash = crypto.createHash("sha256").update(Buffer.from(bytes)).digest("hex");
  const clave = `redes/facebook-stories/${id}/canvas-v1-${hash.slice(0, 24)}.jpg`;
  if (!(await env.PRODUCTOS_R2.head(clave))) {
    await env.PRODUCTOS_R2.put(clave, bytes, {
      httpMetadata: {
        contentType: "image/jpeg",
        cacheControl: "public, max-age=31536000, immutable",
      },
    });
  }
  return Response.json({
    ok: true,
    imagenUrl: `${URL_R2_STORIES}/${clave}`,
  }, { headers: { "Cache-Control": "no-store" } });
}

async function pruebaMongoDB(env) {
  sincronizarEntorno(env);
  if (!env.MONGODB_URI) return Response.json({ ok:false, error:"MONGODB_URI no está configurado" }, { status:500 });
  try {
    const { MongoClient } = await import("mongodb");
    const cliente = new MongoClient(env.MONGODB_URI);
    await cliente.connect();
    await cliente.db("andres_imprimibles").command({ ping:1 });
    await cliente.close();
    return Response.json({ ok:true, mensaje:"Cloudflare conectado correctamente a MongoDB" });
  } catch (error) {
    return Response.json({ ok:false, error:error?.message || "Error conectando con MongoDB" }, { status:500 });
  }
}

function obtenerCookiesWorker(request) {
  const cookies = {};
  const header =
    request.headers.get("cookie") || "";

  header.split(";").forEach((cookie) => {
    const [nombre, ...valor] =
      cookie.trim().split("=");

    if (nombre) {
      cookies[nombre] =
        decodeURIComponent(
          valor.join("=")
        );
    }
  });

  return cookies;
}

function adminAutorizadoWorker(
  request,
  env
) {
  const adminPassword =
    env.ADMIN_PASSWORD;

  if (!adminPassword) {
    return false;
  }

  const tokenEsperado = crypto
    .createHmac(
      "sha256",
      String(adminPassword)
    )
    .update(
      "andres-imprimibles-admin"
    )
    .digest("hex");

  const token =
    obtenerCookiesWorker(request)
      .admin_token;

  if (!token) {
    return false;
  }

  const a = Buffer.from(token);
  const b = Buffer.from(
    tokenEsperado
  );

  if (a.length !== b.length) {
    return false;
  }

  return crypto.timingSafeEqual(
    a,
    b
  );
}

      if (nuevos.length > 1) setErrorEditor("Para este formato se admite un solo archivo.");
      actualizar({ archivos: nuevos.slice(0, 1) });
    }
    setMediaVista(0);
  };
  const quitarArchivo = indice => {
    if (pedidoCreado || enviando) return;
    actualizar({ archivos: item.archivos.filter((_, i) => i !== indice) });
    setMediaVista(0);
    setErrorEditor("");
  };
  const alternarRed = red => {
    if (pedidoCreado || enviando || modoRenovacion) return;
    setRedes(actual => actual.includes(red)
      ? (actual.length === 1 ? actual : actual.filter(x => x !== red))
      : ["instagram", "facebook"].filter(x => x === red || actual.includes(x)));
  };
  const cambiarPais = pais => {
    setPaisZona(pais);
    setZonaHoraria(ZONAS_DISPONIBLES.find(item => item.pais === pais)?.zonas[0]?.[0] || "America/Argentina/Buenos_Aires");
  };

  async function programar(evento) {
    evento.preventDefault();
    if (envioEnCurso.current || !habilitada || !datosCompletos) return;
    if (!redes.length) { setErrorEditor("Elegí al menos una red."); return; }
    if (contenidos.some(c => c.archivos.length < (c.tipo === "carrusel" ? 2 : 1))) {
      setErrorEditor("Elegí los archivos de los cuatro contenidos. Cada carrusel necesita al menos dos imágenes."); return;
    }
    envioEnCurso.current = true;
    setEnviando(true);
    setErrorEditor("");
    setSubida(null);
    let id = pedidoCreado;
    try {
      const publicaciones = contenidos.map(c => ({
        id: c.id, tipo: c.tipo, formato: c.formato, dia: c.dia, hora: c.hora,
        descripcion: c.descripcion, ctaInstagram: c.ctaInstagram, ctaFacebook: c.ctaFacebook,
        urlFacebook: c.urlFacebook, hashtags: c.hashtags,
        archivosSeleccionados: c.archivos.map(a => ({ nombre: a.name, tipo: a.type, tamano: a.size })),
      }));
      if (modoRenovacion) {
        if (!id) {
          const res = await fetch("/api/servicios/publicaciones/renovacion-contenidos/configurar", {
            method: "POST", credentials: "same-origin", cache: "no-store",
            headers: { "Content-Type": "application/json", "X-Comunidad-Panel": "1" },
            body: JSON.stringify({ id: pedidoRenovacion.id, zonaHoraria, publicaciones }),
          });
          const info = await res.json().catch(() => ({}));
          if (!res.ok || info.ok !== true || info.estado !== "en_preparacion") throw new Error(info.error || "No se pudo guardar la semana 2.");
          id = pedidoRenovacion.id;
          setPedidoCreado(id);
        }
        // Tres cargas simultáneas como máximo: reduce las esperas de red/MongoDB
        // sin abrir decenas de conexiones ni saturar un celular.
        const archivos = contenidos.flatMap(c => c.archivos.map((archivo, pos) => ({
          contenido: c.id, pos, archivo,
          clave: `${id}:${c.id}:${pos}:${archivo.name}:${archivo.size}:${archivo.lastModified}`,
        })));
        const pendientes = archivos.filter(a => !subidosSemana2.current.has(a.clave));
        let hechos = archivos.length - pendientes.length;
        let siguiente = 0;
        let primerError = null;
        const subirSiguiente = async () => {
          while (!primerError) {
            const a = pendientes[siguiente++];
            if (!a) return;
            try {
              const ruta = `/api/servicios/publicaciones/renovacion-contenidos/medio?id=${encodeURIComponent(id)}&contenido=${encodeURIComponent(a.contenido)}&pos=${a.pos}`;
              const res = await fetch(ruta, {
                method: "POST", credentials: "same-origin", cache: "no-store",
                headers: { "Content-Type": a.archivo.type, "X-Comunidad-Panel": "1" },
                body: a.archivo,
              });
              const info = await res.json().catch(() => ({}));
              if (!res.ok || info.ok !== true || info.contenido !== a.contenido || info.pos !== a.pos) {
                throw new Error(info.error || `No se pudo subir ${a.archivo.name}.`);
              }
              subidosSemana2.current.add(a.clave);
              hechos++;
              setSubida({ archivo: `${hechos} de ${archivos.length} archivos`,
                porcentaje: Math.round(hechos / archivos.length * 100) });
            } catch (error) {
              // Las cargas ya iniciadas terminan; no se lanzan otras si falla una.
              primerError = primerError || error;
            }
          }
        };
        await Promise.all(Array.from({ length: Math.min(2, pendientes.length) }, () => subirSiguiente()));
        if (primerError) throw primerError;
        const res = await fetch("/api/servicios/publicaciones/renovacion-contenidos/enviar", {
          method:"POST",credentials:"same-origin",cache:"no-store",
          headers:{"Content-Type":"application/json","X-Comunidad-Panel":"1"},
          body:JSON.stringify({id}),
        });
        const info = await res.json().catch(() => ({}));
        if (!res.ok || info.ok !== true || info.estado !== "en_revision") throw new Error(info.error || "No se pudo enviar la semana.");
      } else {
        if (!id) {
          const respuesta = await fetch("/api/servicios/publicaciones/prueba", {
            method: "POST", credentials: "same-origin", cache: "no-store",
            headers: { "Content-Type": "application/json", "X-Comunidad-Panel": "1" },
            body: JSON.stringify({ planId: "comunidad-semanal", moneda: "ARS", redes,
              zonaHoraria, negocio: { nombre, contacto: perfil.nombreApellido, email,
                whatsapp: perfil.whatsapp, url: perfil.web || "", rubro: "", publico: "", tono: "cercano" },
              publicaciones }),
          });
          const datos = await respuesta.json().catch(() => ({}));
          if (!respuesta.ok || !datos.ok || !/^SER-[a-f0-9-]{36}$/i.test(datos.pedidoId || "")) {
            throw new Error(datos.error || "No se pudo registrar la semana gratuita.");
          }
          id = datos.pedidoId;
          setPedidoCreado(id);
        }
        await subirArchivosSemana(id, contenidos, setSubida);
      }
      alFinalizar();
    } catch (error) {
      setSubida(null);
      setErrorEditor(`${error.message || "No se pudieron subir los archivos."} ${id ? "Podés reintentar sobre el mismo pedido." : "Revisá el panel antes de crear otro pedido."}`);
    } finally {
      envioEnCurso.current = false;
      setEnviando(false);
    }
  }

  async function finalizarEnvioAnterior() {
    if (!modoRenovacion || !pedidoRenovacion?.id || envioEnCurso.current) return;
    envioEnCurso.current = true;
    setEnviando(true);
    setErrorEditor("");
    try {
      // Solo completa una renovación real ya existente: no crea otro pedido,
      // no cambia sus horarios y no vuelve a subir archivos a R2.
      const respuesta = await fetch("/api/servicios/publicaciones/renovacion-contenidos/enviar", {
        method: "POST", credentials: "same-origin", cache: "no-store",
        headers: { "Content-Type": "application/json", "X-Comunidad-Panel": "1" },
        body: JSON.stringify({ id: pedidoRenovacion.id }),
      });
      const resultado = await respuesta.json().catch(() => ({}));
      if (!respuesta.ok || resultado.ok !== true || resultado.estado !== "en_revision") {
        throw new Error(resultado.error || "No se pudo finalizar el envío anterior. Revisá si faltan archivos.");
      }
      alFinalizar();
    } catch (error) {
      setErrorEditor(error.message || "No se pudo finalizar el envío anterior.");
    } finally {
      envioEnCurso.current = false;
      setEnviando(false);
    }
  }

  const nombreVisible = nombre || "Tu emprendimiento";
  const usuarioVisible = nombreVisible.toLowerCase().normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]+/g, "_").replace(/^_|_$/g, "") || "tu_emprendimiento";
  const ctaVisible = redPrevia === "instagram" ? item.ctaInstagram : item.ctaFacebook;
  const textoVistaPrevia = (
    <div className="px-3 py-3">
      <p className="whitespace-pre-wrap break-words text-sm font-normal leading-5 text-slate-700">
        {item.descripcion.trim() || "Acá vas a ver la idea de tu publicación."}
      </p>
      {ctaVisible.trim() && <p className="mt-2 text-sm font-medium text-slate-800">{ctaVisible}</p>}
      {redPrevia === "facebook" && item.urlFacebook.trim() &&
        <p className="mt-1 break-all text-sm text-orange-700 underline underline-offset-2">{item.urlFacebook}</p>}
      {item.hashtags.trim() && <p className="mt-2 break-words text-sm text-orange-700">{item.hashtags}</p>}
    </div>
  );
  const sinEdicion = !!pedidoCreado || enviando;
  const TIPOS_EDITOR = {
    reel: { titulo: "Reel", icono: Film },
    carrusel: { titulo: "Carrusel", icono: Images },
    imagen: { titulo: "Imagen", icono: FileImage },
  };

  return <form onSubmit={programar} className="space-y-4 font-normal">
    <div className={`${CARD} p-4 sm:p-5`}>
      <h2 className="text-sm font-medium">{modoRenovacion ? `Programar nueva semana · Semana ${pedidoRenovacion.numeroSemana || 2}` : "Programá tu primera semana"}</h2>
      <p className="mt-1 text-xs leading-5 text-slate-500">{modoRenovacion ? "Configurá los cuatro contenidos de tu semana pagada. Se enviarán a revisión administrativa antes de publicarse." : "Prepará tu contenido semanal y envialo para que lo revisemos y publiquemos en tus cuentas."}</p>
      <div className="mt-4 grid gap-4 sm:grid-cols-2">
        <div>
          <p className="mb-2 text-xs font-medium">Facebook e Instagram</p>
          <div className="grid grid-cols-2 gap-2" role="group" aria-label="Redes para la primera semana">
            {["instagram", "facebook"].map(red => {
              const activa = redes.includes(red);
              return <button type="button" key={red} aria-pressed={activa} disabled={sinEdicion || modoRenovacion}
                onClick={() => alternarRed(red)}
                className={`flex min-h-9 items-center justify-center gap-1.5 rounded-md border px-2 py-1.5 text-xs font-medium transition ${activa ? "border-orange-500 bg-orange-50 text-orange-800" : "border-slate-200 bg-white text-slate-700 hover:border-slate-400"}`}>
                <LogoRed red={red} size={19} />{redLabel(red)} {activa && <Check size={13} />}
              </button>;
            })}
          </div>
          <p className="mt-2 text-[11px] text-slate-500">{cantidad} publicaciones · {redes.length} {redes.length === 1 ? "red" : "redes"}</p>
        </div>
        <div>
          <label className="mb-2 block text-xs font-medium" htmlFor="pais-zona">Zona horaria con selección de país</label>
          <div className="grid grid-cols-2 gap-2">
            <select id="pais-zona" className={CAMPO} value={paisZona} disabled={sinEdicion} onChange={e => cambiarPais(e.target.value)}>
              {ZONAS_DISPONIBLES.map(p => <option key={p.pais} value={p.pais}>{p.pais}</option>)}
            </select>
            <select aria-label="Zona horaria" className={CAMPO} value={zonaHoraria} disabled={sinEdicion} onChange={e => setZonaHoraria(e.target.value)}>
              {(ZONAS_DISPONIBLES.find(p => p.pais === paisZona)?.zonas || []).map(([id, ciudad]) => <option key={id} value={id}>{ciudad}</option>)}
            </select>
          </div>
        </div>
      </div>
    </div>

    <div className={`${CARD} p-4 sm:p-5`}>
      <h2 className="text-sm font-medium">Elegí el contenido que querés publicar</h2>
      <p className="mt-1 text-xs leading-5 text-slate-500">Cada contenido conserva sus propios archivos, formato, textos, día, horario y hagstags.</p>
    </div>

    <div className="grid items-start gap-4 lg:grid-cols-[minmax(0,1fr)_350px]">
      <div className="min-w-0 space-y-4">
        <details open className={`${CARD} group`}>
          <summary className="flex cursor-pointer list-none items-center justify-between gap-2 px-4 py-3 text-xs font-medium [&::-webkit-details-marker]:hidden">
            Contenidos <ChevronDown size={14} className="transition group-open:rotate-180" />
          </summary>
          <div className="border-t border-slate-100 p-4">
            <p className="mb-3 text-xs text-slate-500">Formato, archivos, día y hora.</p>
            <div className="grid grid-cols-2 gap-2 sm:grid-cols-4" role="group" aria-label="Elegir contenido">
              {contenidos.map((c, i) => {
                const Icono = TIPOS_EDITOR[c.tipo].icono;
                return <button type="button" key={c.id} aria-pressed={seleccionado === i}
                  onClick={() => { setSeleccionado(i); setMediaVista(0); setErrorEditor(""); }}
                  className={`flex items-center justify-center gap-1.5 rounded-md border px-2 py-2 text-xs font-medium transition ${seleccionado === i ? "border-orange-500 bg-orange-50 text-orange-800" : "border-slate-200 bg-white text-slate-700 hover:border-slate-400"}`}>
                  <Icono size={15} /> Contenido {i + 1}
                </button>;
              })}
            </div>
            <div className="mt-4 grid gap-3 sm:grid-cols-2">
              <label className="text-xs font-medium">Tipo de publicación
                <select className={`mt-1 ${CAMPO}`} value={item.tipo} disabled={sinEdicion} onChange={e => cambiarTipo(e.target.value)}>
                  <option value="imagen">Imagen</option><option value="carrusel">Carrusel</option><option value="reel">Reel</option>
                </select>
              </label>
              <label className="text-xs font-medium">Formato
                <select className={`mt-1 ${CAMPO}`} value={item.formato} disabled={sinEdicion} onChange={e => actualizar({ formato: e.target.value })}>
                  {(item.tipo === "reel" ? ["9:16"] : ["4:5", "1:1"]).map(x => <option key={x} value={x}>{x}</option>)}
                </select>
              </label>
              <label className="text-xs font-medium">Día
                <select className={`mt-1 ${CAMPO}`} value={item.dia} disabled={sinEdicion} onChange={e => actualizar({ dia: e.target.value })}>
                  {DIAS_SEMANA.map(x => <option key={x}>{x}</option>)}
                </select>
              </label>
              <label className="text-xs font-medium">Horario local
                <select className={`mt-1 ${CAMPO}`} value={item.hora} disabled={sinEdicion} onChange={e => actualizar({ hora: e.target.value })}>
                  {HORAS_SEMANA.map(x => <option key={x}>{x}</option>)}
                </select>
              </label>
            </div>
            <div className="mt-4 border-t border-slate-100 pt-4">
              <p className="mb-2 text-xs font-medium">{item.tipo === "reel" ? "Video del reel" : esCarrusel ? "Imágenes del carrusel" : "Imagen de la publicación"}</p>
              <label className={`flex items-center justify-center gap-2 rounded-md border border-dashed border-slate-300 bg-white px-3 py-3 text-center transition ${sinEdicion ? "cursor-not-allowed opacity-60" : "cursor-pointer hover:border-orange-500"}`}>
                <CloudUpload size={19} className="shrink-0 text-orange-700" />
                <span className="text-xs font-medium">{item.archivos.length ? "Agregar o cambiar archivos" : "Elegir archivos"}</span>
                <span className="hidden text-xs text-slate-500 sm:inline">{modoRenovacion ? "Hasta 25 MB por archivo" : item.tipo === "reel" ? "MP4 · 150 MB" : "JPG, PNG, WebP · 20 MB"}</span>
                <input type="file" className="sr-only" disabled={sinEdicion}
                  accept={item.tipo === "reel" ? "video/mp4,.mp4" : "image/jpeg,image/png,image/webp,.jpg,.jpeg,.png,.webp"}
                  multiple={esCarrusel} onChange={seleccionarArchivos} />
              </label>
              <p className="mt-2 text-[11px] text-slate-500">{esCarrusel ? "De 2 a 10 imágenes" : item.tipo === "reel" ? "Un video MP4" : "Una imagen"}.</p>
              {modoRenovacion && redes.includes("instagram") && item.tipo !== "reel" && (
                <p className="mt-1 text-[11px] text-slate-600">
                  Admitimos JPG, PNG y WebP. Durante la revisión, preparamos una copia JPEG antes de programar Instagram.
                </p>
              )}
              {item.archivos.length > 0 && <div className="mt-2 flex flex-wrap gap-1.5">
                {item.archivos.map((archivo, indice) => <span key={`${item.id}-${indice}-${archivo.name}`} className="inline-flex max-w-full items-center gap-1.5 rounded-md border border-slate-200 px-2 py-1 text-xs">
                  <span className="max-w-36 truncate" title={archivo.name}>{archivo.name}</span>
                  <button type="button" aria-label={`Quitar ${archivo.name}`} disabled={sinEdicion} className="text-slate-500 hover:text-red-700 disabled:opacity-50"
                    onClick={() => quitarArchivo(indice)}><Trash2 size={13} /></button>
                </span>)}
              </div>}
              {errorEditor && <p className="mt-2 text-xs text-red-700" role="alert">{errorEditor}</p>}
            </div>
          </div>
        </details>

        <details className={`${CARD} group`}>
          <summary className="flex cursor-pointer list-none items-center justify-between gap-2 px-4 py-3 text-xs font-medium [&::-webkit-details-marker]:hidden">
            Texto y detalles de la publicación <ChevronDown size={14} className="transition group-open:rotate-180" />
          </summary>
          <div className="grid gap-3 border-t border-slate-100 p-4 sm:grid-cols-2">
            <p className="col-span-full text-xs text-slate-500">Mensaje, llamadas a la acción, enlaces y hashtags.</p>
            <label className="col-span-full text-xs font-medium">Copy o texto de la publicación
              <textarea className={`${CAMPO} mt-1 min-h-24`} placeholder="Escribí el texto de tu publicación" maxLength={1500} disabled={sinEdicion} value={item.descripcion} onChange={e => actualizar({ descripcion: e.target.value })} />
            </label>
            {redes.includes("instagram") && <label className="text-xs font-medium">CTA para Instagram
              <input className={`mt-1 ${CAMPO}`} placeholder="ej: Enlace en nuestra bio" maxLength={100} value={item.ctaInstagram} disabled={sinEdicion} onChange={e => actualizar({ ctaInstagram: e.target.value })} />
            </label>}
            {redes.includes("facebook") && <>
              <label className="text-xs font-medium">CTA para Facebook
                <input className={`mt-1 ${CAMPO}`} placeholder="ej: Visitá nuestra tienda" maxLength={100} value={item.ctaFacebook} disabled={sinEdicion} onChange={e => actualizar({ ctaFacebook: e.target.value })} />
              </label>
              <label className="text-xs font-medium">URL para Facebook
                <input className={`mt-1 ${CAMPO}`} placeholder="ej: minegocio.com.ar" maxLength={500} value={item.urlFacebook} disabled={sinEdicion} onChange={e => actualizar({ urlFacebook: e.target.value })} />
              </label>
            </>}
            <label className="text-xs font-medium">Hashtags
              <input className={`mt-1 ${CAMPO}`} placeholder="#redesenpiloto #minegocio #redes" maxLength={240} value={item.hashtags} disabled={sinEdicion} onChange={e => actualizar({ hashtags: e.target.value })} />
            </label>
          </div>
        </details>
      </div>

      <aside className="min-w-0 lg:sticky lg:top-20" aria-label="Vista previa de la publicación">
        <details open className={`${CARD} group`}>
          <summary className="flex cursor-pointer list-none items-center justify-between gap-2 px-4 py-3 text-xs font-medium [&::-webkit-details-marker]:hidden">
            Vista previa <ChevronDown size={14} className="transition group-open:rotate-180" />
          </summary>
          <div className="border-t border-slate-100 p-3 sm:p-4">
            <p className="mb-3 text-xs text-slate-500">Contenido {seleccionado + 1} · Así se verá en cada red</p>
            <div className="mb-3 flex gap-2" role="group" aria-label="Ver publicación por red">
              {["instagram", "facebook"].filter(red => redes.includes(red)).map(red => <button type="button" key={red} aria-pressed={redPrevia === red} onClick={() => setRedVista(red)}
                className={`flex min-h-9 flex-1 items-center justify-center gap-1.5 rounded-md border px-2 py-1 text-xs font-medium ${redPrevia === red ? "border-orange-500 bg-orange-50 text-orange-800" : "border-slate-200 text-slate-600"}`}>
                <LogoRed red={red} size={19} /> {redLabel(red)}
              </button>)}
            </div>
            <div className="overflow-hidden rounded-md border border-slate-200 bg-white">
              <div className="flex items-center gap-2 px-3 py-3">
                <LogoRed red={redPrevia} size={27} />
                <div className="min-w-0 flex-1">
                  <p className="truncate text-xs font-medium text-slate-900">{redPrevia === "instagram" ? usuarioVisible : nombreVisible}</p>
                  <p className="text-[11px] font-normal text-slate-500">{redPrevia === "instagram" ? "Publicación de Instagram" : "Página de Facebook · Público"}</p>
                </div>
                <MoreHorizontal size={19} className="text-slate-500" />
              </div>
              {redPrevia === "facebook" && textoVistaPrevia}
              <div className="bg-slate-50 px-3 py-2">
                <div className={`relative mx-auto overflow-hidden rounded-md bg-slate-100 ${item.formato === "9:16" ? "w-full max-w-[225px]" : "w-full max-w-[300px]"}`}
                  style={{ aspectRatio: item.formato.replace(":", " / ") }}>
                  <VistaArchivoLocal archivo={item.archivos[mediaVista] || item.archivos[0]} tipo={item.tipo} />
                  {esCarrusel && item.archivos.length > 1 && <>
                    <button type="button" aria-label="Imagen anterior" onClick={() => setMediaVista(i => (i - 1 + item.archivos.length) % item.archivos.length)}
                      className="absolute left-1.5 top-1/2 -translate-y-1/2 rounded-md bg-white/90 p-1.5 text-slate-800"><ChevronLeft size={16} /></button>
                    <button type="button" aria-label="Imagen siguiente" onClick={() => setMediaVista(i => (i + 1) % item.archivos.length)}
                      className="absolute right-1.5 top-1/2 -translate-y-1/2 rounded-md bg-white/90 p-1.5 text-slate-800"><ChevronRight size={16} /></button>
                    <span className="absolute right-1.5 top-1.5 rounded-md bg-black/65 px-2 py-1 text-[10px] text-white">{mediaVista + 1}/{item.archivos.length}</span>
                  </>}
                </div>
              </div>
              {redPrevia === "instagram" ? <div className="flex items-center gap-3 px-3 pt-3 text-slate-800" aria-label="Acciones orientativas de Instagram">
                <Heart size={19} /><MessageCircle size={19} /><Send size={19} />
              </div> : <div className="flex items-center justify-around border-b border-slate-100 px-3 py-2 text-xs text-slate-500" aria-label="Acciones orientativas de Facebook">
                <span className="inline-flex items-center gap-1"><ThumbsUp size={15} /> Me gusta</span>
                <span className="inline-flex items-center gap-1"><MessageCircle size={15} /> Comentar</span>
                <span className="inline-flex items-center gap-1"><Send size={15} /> Compartir</span>
              </div>}
              {redPrevia === "instagram" && textoVistaPrevia}
            </div>
            <p className="mt-2 text-xs font-normal leading-4 text-slate-500">Simulación orientativa: la apariencia final puede variar según la red social.</p>
          </div>
        </details>
      </aside>
    </div>

    <details open className={`${CARD} group`}>
      <summary className="flex cursor-pointer list-none items-center justify-between gap-2 px-4 py-3 text-xs font-medium [&::-webkit-details-marker]:hidden">
        Resumen y programación <ChevronDown size={14} className="transition group-open:rotate-180" />
      </summary>

        <div><p className="font-medium">{cantidad} publicaciones</p><p className="mt-0.5 text-slate-500">{redes.map(redLabel).join(" · ")}</p></div>

function RenovacionPaypalReal({pedidoId,seleccion,alActualizar}) {
  const [info,setInfo]=useState(null);
  const [ocupado,setOcupado]=useState(false);
  const [error,setError]=useState("");
  const [mensaje,setMensaje]=useState("");
  const API="/api/servicios/publicaciones/renovacion-paypal";
  const consultar=useCallback(async()=>{
    const r=await fetch(`${API}?pedidoId=${encodeURIComponent(pedidoId)}`,{credentials:"same-origin",cache:"no-store"});
    const j=await r.json().catch(()=>null);
    if(!r.ok||j?.ok!==true)throw Error(j?.error||`No se pudo consultar PayPal (HTTP ${r.status}).`);
    setInfo(j);return j;
  },[pedidoId]);
  useEffect(()=>{let activo=true;consultar().catch(e=>{if(activo)setError(e.message)});return()=>{activo=false;};},[consultar]);
  const crear=async()=>{
    setOcupado(true);setError("");setMensaje("");
    try {
      const redes=seleccion==="ambas"?["facebook","instagram"]:[seleccion];
      const r=await fetch(`${API}/crear`,{method:"POST",credentials:"same-origin",cache:"no-store",
        headers:{"Content-Type":"application/json","X-Comunidad-Panel":"1"},body:JSON.stringify({pedidoId,redes})});
      const j=await r.json().catch(()=>null);
      if(!r.ok||!j?.ok)throw Error(j?.error||`Error HTTP ${r.status} al preparar PayPal.`);
      if(j.pagado){await consultar();setMensaje("Esta semana ya tiene el pago confirmado.");return;}
      if(typeof j.url!=="string")throw Error("No se recibió la página de PayPal.");
      // PayPal recibe al usuario directamente; no se crean ventanas emergentes.
      window.location.assign(j.url);
    }catch(e){setError(e.message||"PayPal no está disponible.");}
    finally{setOcupado(false);}
  };
  const verificar=async()=>{
    if(!info?.pedido?.id)return;
    setOcupado(true);setError("");setMensaje("");
    try{
      const r=await fetch(`${API}/capturar`,{method:"POST",credentials:"same-origin",cache:"no-store",
        headers:{"Content-Type":"application/json","X-Comunidad-Panel":"1"},
        body:JSON.stringify({pedidoId,semanaId:info.pedido.id})});
      const j=await r.json().catch(()=>null);
      if(!r.ok||!j?.ok)throw Error(j?.error||`PayPal aún no confirmó el pago (HTTP ${r.status}).`);
      setMensaje("Pago confirmado. La activación administrativa de la semana continúa por separado.");
      alActualizar?.(j.pedido||null);await consultar();
    }catch(e){setError(e.message||"No se pudo verificar el pago.");}
    finally{setOcupado(false);}
  };
  const pedido=info?.pedido,precio=seleccion==="ambas"?"15.00":"10.00";
  const finalizada=pedido&&(pedido.configuracionEstado==="programada"||pedido.estadoSemana==="finalizada");
  const pendiente=pedido&&!finalizada;
  const otrasRedes=pendiente&&JSON.stringify([...(pedido.redes||[])].sort())!==JSON.stringify((seleccion==="ambas"?["facebook","instagram"]:[seleccion]).sort());
  return <div className="space-y-3 rounded-md border border-slate-200 bg-white p-3 text-xs">
    <p className="font-medium text-slate-900">PayPal · semana adicional de 7 días</p>
    <p className="text-slate-600">Importe seleccionado: <strong>USD {precio}</strong>. El importe definitivo se verifica en el servidor antes de enviar a PayPal.</p>
    {info?.entorno==="sandbox"&&<p className="text-amber-700">Entorno de prueba Sandbox: no se cobran fondos reales.</p>}
    {info&&!info.habilitada&&<p className="text-amber-700">{info.motivo||"PayPal todavía no está habilitado."}</p>}
    {info?.bloqueada&&<p className="text-amber-700">{info.bloqueada}</p>}
    {pedido&&<div className="rounded-md bg-slate-50 p-3 text-slate-700">
      <p>Semana {pedido.numeroSemana} · Estado: <strong>{pedido.estado}</strong></p>
      <p className="mt-1">Redes reservadas: {(pedido.redes||[]).map(redLabel).join(" · ")}</p>
      <p className="mt-1">Importe reservado: <strong>USD {Number(pedido.importe).toFixed(2)}</strong></p>
      {pedido.pagoConfirmado&&<p className="mt-1 text-emerald-700">Pago confirmado. Pendiente de la preparación o programación administrativa.</p>}
    </div>}
    {otrasRedes&&<p className="text-amber-700">La semana pendiente tiene otras redes seleccionadas. Terminá ese pago antes de cambiar las redes.</p>}
    {info?.habilitada&&!info.bloqueada&&(!pendiente||(!pedido?.pagoConfirmado&&!otrasRedes))&&
      <button type="button" disabled={ocupado} className={BTN_PRIMARIO} onClick={crear}>
        {ocupado?"Preparando PayPal...":pendiente?"Continuar pago pendiente":finalizada?`Solicitar Semana ${(pedido.numeroSemana||2)+1} con PayPal`:"Pagar con PayPal"}
      </button>}
    {info?.habilitada&&pendiente&&!pedido.pagoConfirmado&&<button type="button" disabled={ocupado} className={BTN_SECUNDARIO} onClick={verificar}>
      {ocupado?"Verificando...":"Verificar si el pago se completó"}
    </button>}
    {error&&<p role="alert" className="text-red-700">{error}</p>}
    {mensaje&&<p role="status" className="text-emerald-700">{mensaje}</p>}
  </div>;
}

function RetornoPaypalRenovacion({alActualizar}) {
  const [resultado,setResultado]=useState("");
  const [error,setError]=useState("");
  const ejecutado=useRef(false);
  useEffect(()=>{
    if(ejecutado.current)return;
    const url=new URL(window.location.href),modo=url.searchParams.get("paypal");
    const pedidoId=url.searchParams.get("pedidoId"),semanaId=url.searchParams.get("semanaId");
    if(!["retorno","cancelado"].includes(modo)||!VALIDAR_PEDIDO.test(pedidoId||"")||!/^renovacion-real:SER-[a-f0-9-]{36}(?::(?:[3-9]|[1-9][0-9]{1,2}))?$/i.test(semanaId||""))return;
    ejecutado.current=true;
    // No conservar tokens de PayPal ni identificadores de pago en la barra de direcciones.
    url.searchParams.delete("paypal");url.searchParams.delete("pedidoId");url.searchParams.delete("semanaId");url.searchParams.delete("token");url.searchParams.delete("PayerID");
    window.history.replaceState(window.history.state,"",url.pathname+(url.search||"")+url.hash);
    if(modo==="cancelado"){setResultado("Cancelaste la autorización en PayPal. No se registró ningún cobro nuevo.");return;}
    (async()=>{
      try{
        const r=await fetch("/api/servicios/publicaciones/renovacion-paypal/capturar",{
          method:"POST",credentials:"same-origin",cache:"no-store",headers:{"Content-Type":"application/json","X-Comunidad-Panel":"1"},
          body:JSON.stringify({pedidoId,semanaId})});
        const j=await r.json().catch(()=>null);
        if(!r.ok||!j?.ok)throw Error(j?.error||`No se pudo verificar el pago (HTTP ${r.status}).`);
        alActualizar?.(j.pedido||null);
        setResultado("PayPal confirmó el pago. La semana sigue pendiente de activación y revisión de contenidos.");
      }catch(e){setError(`${e.message} Si el cargo figura en PayPal, no repitas el pago: abrí la opción PayPal y utilizá «Verificar si el pago se completó».`);}
    })();
  },[alActualizar]);
  if(!resultado&&!error)return null;
  return <div className={`mb-4 rounded-md border p-3 text-sm ${error?"border-red-200 bg-red-50 text-red-800":"border-green-200 bg-green-50 text-green-800"}`} role="status">{error||resultado}</div>;
}

function RenovacionTransferenciaReal({ pedidoId, seleccion, alActualizar }) {

  const [archivoRenovacion, setArchivoRenovacion] = useState(null);
  const [ocupadoRenovacion, setOcupadoRenovacion] = useState(false);
  const [errorRenovacion, setErrorRenovacion] = useState("");
  const [mensajeRenovacion, setMensajeRenovacion] = useState("");
  const [avisoRecibido, setAvisoRecibido] = useState(false);
  const API_RENOVACION = "/api/servicios/publicaciones/renovacion-transferencia";
  const consultar = useCallback(async () => {
    const respuesta = await fetch(`${API_RENOVACION}?pedidoId=${encodeURIComponent(pedidoId)}`, {
      credentials: "same-origin", cache: "no-store",
    });
    const tipo = respuesta.headers.get("content-type") || "";
    if (!tipo.includes("application/json")) throw new Error(`No se pudo consultar el cobro (HTTP ${respuesta.status}).`);
    const resultado = await respuesta.json();
    if (!respuesta.ok || !resultado.ok) throw new Error(resultado.error || "No se pudo consultar la renovación.");
    setDatosRenovacion(resultado);
    alActualizar?.(resultado.pedido || null);
    return resultado;
  }, [pedidoId]);
  useEffect(() => {
    let activo = true;
    (async () => {
      try {
        const respuesta = await fetch(`${API_RENOVACION}?pedidoId=${encodeURIComponent(pedidoId)}`, {
          credentials: "same-origin", cache: "no-store",
        });
        const tipo = respuesta.headers.get("content-type") || "";
        if (!tipo.includes("application/json")) throw new Error(`La API devolvió HTML (HTTP ${respuesta.status}).`);
        const resultado = await respuesta.json();
        if (!respuesta.ok || !resultado.ok) throw new Error(resultado.error || "No se pudo consultar el pago.");
        if (activo) setDatosRenovacion(resultado);
      } catch (err) { if (activo) setErrorRenovacion(err.message || "No se pudo consultar el pago."); }
    })();
    return () => { activo = false; };
  }, [pedidoId]);
  const crear = async () => {
    setOcupadoRenovacion(true); setErrorRenovacion(""); setMensajeRenovacion("");
    const redes = seleccion === "ambas" ? ["facebook", "instagram"] : [seleccion];
    try {
      const respuesta = await fetch(`${API_RENOVACION}/crear`, {
        method: "POST", credentials: "same-origin", cache: "no-store",
        headers: { "Content-Type": "application/json", "X-Comunidad-Panel": "1" },
        body: JSON.stringify({ pedidoId, redes }),
      });
      const tipo = respuesta.headers.get("content-type") || "";
      if (!tipo.includes("application/json")) throw new Error(`El servidor devolvió una página (HTTP ${respuesta.status}).`);
      const json = await respuesta.json();
      if (!respuesta.ok || !json.ok) throw new Error(json.error || "No se pudo preparar la renovación.");
      await consultar();
    } catch (err) { setErrorRenovacion(err.message || "No se pudo preparar la renovación."); }
    finally { setOcupadoRenovacion(false); }
  };
  const subir = async () => {
    if (!archivoRenovacion || !datosRenovacion?.pedido) return;
    if (!["image/jpeg", "image/png", "application/pdf"].includes(archivoRenovacion.type) ||
        archivoRenovacion.size > 8 * 1024 * 1024 || archivoRenovacion.size < 100) {
      setErrorRenovacion("Seleccioná un comprobante JPG, PNG o PDF de hasta 8 MB."); return;
    }
    setOcupadoRenovacion(true); setErrorRenovacion(""); setMensajeRenovacion("");
    try {
      const respuesta = await fetch(`${API_RENOVACION}/comprobante?pedidoId=${encodeURIComponent(pedidoId)}`, {
        method: "POST", credentials: "same-origin", cache: "no-store",
        headers: { "Content-Type": archivoRenovacion.type, "X-Comunidad-Panel": "1" },
        body: archivoRenovacion,
      });
      const tipo = respuesta.headers.get("content-type") || "";
      if (!tipo.includes("application/json")) throw new Error(`El servidor devolvió una página (HTTP ${respuesta.status}). Revisá los registros de Cloudflare.`);
      const json = await respuesta.json();
      if (!respuesta.ok || !json.ok) throw new Error(json.error || `No se pudo enviar el comprobante (HTTP ${respuesta.status}).`);
      setArchivoRenovacion(null);
      await consultar();
      setMensajeRenovacion("Comprobante recibido. Estamos verificando tu transferencia y te avisaremos por correo cuando se apruebe.");
      setAvisoRecibido(true);
    } catch (err) { setErrorRenovacion(err.message || "No pudimos subir el comprobante."); }
    finally { setOcupadoRenovacion(false); }
  };
  const copiar = async dato => {
    try { await navigator.clipboard.writeText(dato); setMensajeRenovacion("Dato copiado."); }
    catch { setErrorRenovacion("Seleccioná el dato y copialo manualmente."); }
  };
  const pedido = datosRenovacion?.pedido;
  if(pedido?.metodoPago==='paypal')return <p className="rounded-md border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800">La semana actual se solicitó por PayPal. Para continuar el pago, seleccioná PayPal en el formulario.</p>;
  const estados = {
    pendiente_comprobante: "Esperando comprobante", en_revision: "Comprobante en revisión",
    aprobado: "Pago aprobado", rechazado: "Comprobante rechazado · podés enviar uno nuevo",
  };
  return <div className="space-y-3 rounded-md border border-slate-200 bg-white p-3 text-xs">
    {avisoRecibido && <div role="dialog" aria-modal="true" aria-label="Comprobante recibido" className="fixed inset-0 z-50 flex items-center justify-center bg-slate-950/40 p-4">
      <div className="w-full max-w-md space-y-4 rounded-md bg-white p-5 text-slate-800 shadow-xl">
        <p className="text-sm font-medium">Comprobante recibido</p>
        <p className="text-xs leading-6">Recibimos tu comprobante y estamos verificando la transferencia. Cuando aprobemos el pago, te enviaremos un correo de confirmación.</p>
        <p className="text-xs text-slate-600">La semana adicional no empieza hasta que su programación sea confirmada.</p>
        <button type="button" className={BTN_PRIMARIO} onClick={() => setAvisoRecibido(false)}>Entendido</button>
      </div>
    </div>}
    <p className="font-medium text-slate-900">Transferencia bancaria · semana adicional de 7 días</p>
    <p className="text-slate-600">Importe: <strong>{pedido ? `$${pedido.importe.toLocaleString("es-AR")}` : seleccion === "ambas" ? "$15.000" : "$10.000"} ARS</strong>. El pago se acredita únicamente después de verificar la transferencia.</p>
    {datosRenovacion === null && !errorRenovacion && <p className="text-slate-500">Consultando disponibilidad...</p>}
    {datosRenovacion && !datosRenovacion.habilitada &&
      <p role="status" className="text-amber-700">{datosRenovacion.motivo || "Los pagos reales todavía no están habilitados."}</p>}
    {datosRenovacion?.habilitada && !pedido &&
      <button type="button" disabled={ocupadoRenovacion} className={BTN_PRIMARIO} onClick={crear}>
        {ocupadoRenovacion ? "Preparando pedido..." : "Continuar con transferencia"}

    {pedido?.configuracionEstado === "programada" && datosRenovacion?.habilitada &&
      <button type="button" disabled={ocupadoRenovacion} className={BTN_PRIMARIO} onClick={crear}>
        {ocupadoRenovacion ? "Preparando renovación..." : `Solicitar Semana ${(pedido.numeroSemana || 2) + 1}`}

        <p className="mt-1">Importe reservado: <strong>${pedido.importe.toLocaleString("es-AR")} ARS</strong></p>

      {["pendiente_comprobante", "rechazado"].includes(pedido.estado) && datosRenovacion?.habilitada && <>

function fechaLocal(valor, zonaHoraria) {
  if (!valor) return "Por confirmar";
  const d = new Date(valor);
  if (!Number.isFinite(d.getTime())) return "Por confirmar";
  try { return new Intl.DateTimeFormat("es-AR", {
    day:"numeric", month:"short", hour:"2-digit", minute:"2-digit",
    ...(zonaHoraria ? { timeZone:zonaHoraria } : {}),
  }).format(d); }
  catch { return new Intl.DateTimeFormat("es-AR", {day:"numeric",month:"short",hour:"2-digit",minute:"2-digit"}).format(d); }
}
function enlacePublicado(contenido, red) {
  // Cada red utiliza exclusivamente su enlace confirmado: un enlace general
  // puede corresponder a Instagram y no debe mostrarse como Facebook.
  const valor = contenido?.urlsPublicadas?.[red] || contenido?.enlaces?.[red] || contenido?.[`${red}UrlPublicada`];
  if (typeof valor !== "string") return "";
  try {
    const u = new URL(valor);
    const dominios = red === "instagram"
      ? ["instagram.com", "www.instagram.com"]
      : ["facebook.com", "www.facebook.com", "m.facebook.com"];
    return u.protocol === "https:" && dominios.includes(u.hostname) ? u.toString() : "";
  } catch { return ""; }
}
function urlWebSegura(valor) {
  if (typeof valor !== "string" || !valor.trim()) return "";
  try {
    const url = new URL(/^[a-z][a-z0-9+.-]*:/i.test(valor) ? valor : `https://${valor}`);
    return ["http:", "https:"].includes(url.protocol) && !url.username && !url.password ? url.toString() : "";
  } catch { return ""; }
}
function horaLocalEn12Horas(hora) {
  const partes = /^(\d{2}):([0-5]\d)$/.exec(hora || "");
  if (!partes) return "Hora por confirmar";
  const h = Number(partes[1]);
  return `${h % 12 || 12}:${partes[2]} ${h < 12 ? "a. m." : "p. m."}`;
}

function fechaContenido(c, zona) {
  const definitiva = c.fechaHoraDefinitiva || c.fechaProgramada || c.programadoPara;
  if (definitiva && Number.isFinite(Date.parse(definitiva))) {
    try {
      return new Intl.DateTimeFormat("es-AR", {
        weekday: "long", day: "numeric", hour: "2-digit", minute: "2-digit",
        ...(zona ? { timeZone: zona } : {}),
      }).format(new Date(definitiva));
    } catch { /* Mantener el horario elegido aunque la zona sea inválida. */ }
  }
  // El cliente ya eligió un día y una hora. No inventar un número de día
  // hasta que Administración confirme la fecha concreta.
  if (c.diaPreferido && c.horaLocalPreferida) {
    return `${c.diaPreferido} · ${c.horaLocalPreferida}`;
  }
  return "Horario pendiente";
}

async function leerJsonAcotado(request) {
  if (!/^application\/json(?:;|$)/i.test(request.headers.get("content-type") || "")) return null;
  const lector = request.body?.getReader();
  if (!lector) return null;
  const decoder = new TextDecoder();
  let contenido = "", total = 0;
  try {
    while (true) {
      const { done, value } = await lector.read();
      if (done) break;
      total += value.byteLength;
      if (total > 2048) { await lector.cancel(); return null; }
      contenido += decoder.decode(value, { stream: true });
    }
    const dato = JSON.parse(contenido + decoder.decode());
    return dato && typeof dato === "object" && !Array.isArray(dato) ? dato : null;
  } catch { return null; }
}

}

    if (autorizado === "no_habilitado") return json({ ok: false, error: "El pedido debe estar habilitado para preparación y tener Facebook seleccionado." }, 409);

    const redirectUri = `${url.origin}${CALLBACK}`;
    const now = new Date();
    await conDb(env, async db => {
      const intentos = db.collection(INTENTOS);
      await intentos.createIndex({ expiraEn: 1 }, { expireAfterSeconds: 0, name: "oauth_expira_ttl" });
      await intentos.insertOne({ _id: hash(state), pedidoId: datos.pedidoId, origen: url.origin, creadoEn: now, expiraEn: new Date(now.getTime() + PLAZO_MS) });
    });
    const destino = new URL(`https://www.facebook.com/${META_VERSION}/dialog/oauth`);
    destino.searchParams.set("client_id", config.appId);
    destino.searchParams.set("redirect_uri", redirectUri);
    destino.searchParams.set("response_type", "code");
    if (config.configId) {
      destino.searchParams.set("config_id", config.configId);
      destino.searchParams.set("override_default_response_type", "true");
    } else {
      destino.searchParams.set("scope", "pages_show_list,pages_read_engagement,pages_manage_posts");
    }
    destino.searchParams.set("state", state);
    return json({ ok: true, authorizationUrl: destino.toString() }, 200, cookieEstado(state));
  } catch {
    return json({ ok: false, error: "No fue posible iniciar la conexión con Meta." }, 503);
  }
}

async function estadoODesconectar(request, env, url, desconectar) {
  if (request.method !== (desconectar ? "POST" : "GET")) {
    return json({ ok: false, error: "Método no permitido." }, 405);
  }
  if (desconectar && request.headers.get("origin") !== url.origin) {
    return json({ ok: false, error: "Origen no permitido." }, 403);
  }
  const datos = desconectar ? await leerJsonAcotado(request) : { pedidoId: url.searchParams.get("pedidoId") };
  if (!pedidoValido(datos?.pedidoId)) return json({ ok: false, error: "Pedido inválido." }, 400);
  try {
    const usuario = await obtenerClienteRegistrado(request, env).catch(() => null);
    return await conDb(env, async db => {
      const pedido = await db.collection("pedidos_servicios").findOne({ pedidoId: datos.pedidoId, tipo: "publicaciones" });
      if (!pedido || !accesoCliente(request, pedido, usuario)) return json({ ok: false, error: "Pedido no encontrado o acceso no autorizado." }, 404);
      if (desconectar) {
        await db.collection(COLECCION).deleteOne({ _id: `facebook:${datos.pedidoId}`, pedidoId: datos.pedidoId });
        // Cancelar también autorizaciones pendientes para impedir nuevas conexiones tardías.
        await db.collection(INTENTOS).deleteMany({ pedidoId: datos.pedidoId });
        return json({ ok: true, desconectada: true });
      }
      const registro = await db.collection(COLECCION).findOne(
        { _id: `facebook:${datos.pedidoId}`, pedidoId: datos.pedidoId },
        { projection: { _id: 0, pageId: 1, nombre: 1, tareas: 1, estado: 1, conectadoEn: 1, actualizadoEn: 1 } },
      );
      // La UI necesita estos indicadores del servidor: una orden de Instagram
      // o un pago pendiente no deben mostrar la opción de conectar Facebook.
      // Se devuelven solo después de validar el Bearer privado del pedido.
      const facebookContratado = Array.isArray(pedido.redes) && pedido.redes.includes("facebook");
      const pagoConfirmado = pedido.estadoPago === "confirmado" && Boolean(pedido.pagadoEn);
      return json({
        ok: true,
        facebookContratado,
        pagoConfirmado,
        pruebaGratuita: pedido.modalidadContratacion === "prueba_gratuita",
        habilitada: pedidoHabilitado(pedido),
        conectada: Boolean(registro),
        conexion: registro || null,
      });
    });
  } catch {
    return json({ ok: false, error: "No fue posible consultar o desconectar Facebook." }, 503);
  }
}

async function callback(request, env, url) {
  if (request.method !== "GET") return pagina("Método no permitido", "<p>El retorno de Meta requiere GET.</p>", 405);
  const state = url.searchParams.get("state") || "";
  if (!compararEstado(state, leerCookie(request))) {
    return pagina("Autorización vencida", "<p>La autorización no corresponde a este navegador o ya venció.</p>", 403);
  }
  const config = configuracion(env);
  if (!config) return pagina("Configuración pendiente", "<p>La configuración de Meta no está disponible.</p>", 503);
  let intento;
  try {
    intento = await conDb(env, db => db.collection(INTENTOS).findOneAndDelete({
      _id: hash(state), origen: url.origin, expiraEn: { $gt: new Date() },
    }));
  } catch {
    return pagina("Conexión no disponible", "<p>No fue posible validar esta autorización.</p>", 503);
  }
  if (!intento) return pagina("Autorización vencida", "<p>La autorización ya se utilizó o venció.</p>", 403);
  if (url.searchParams.has("error")) return pagina("Autorización cancelada", "<p>Meta no concedió el acceso.</p>", 400);
  const code = url.searchParams.get("code") || "";
  if (!code || code.length > 4000) return pagina("Autorización inválida", "<p>Falta el código de Meta.</p>", 400);
  try {
    const sigueHabilitado = await conDb(env, async db => {
      const pedido = await db.collection("pedidos_servicios").findOne({ pedidoId: intento.pedidoId, tipo: "publicaciones" });
      return pedidoHabilitado(pedido);
    });
    if (!sigueHabilitado) return pagina("Pedido no habilitado", "<p>El pedido ya no admite conectar Facebook.</p>", 409);
    const redirectUri = `${url.origin}${CALLBACK}`;
    const datos = await respuestaMeta(await fetch(`https://graph.facebook.com/${META_VERSION}/oauth/access_token`, {
      method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({ client_id: config.appId, client_secret: config.secret, redirect_uri: redirectUri, code }),
    }));
    if (typeof datos.access_token !== "string") throw new Error("TOKEN_INVALIDO");
    const extendido = await respuestaMeta(await fetch(`https://graph.facebook.com/${META_VERSION}/oauth/access_token`, {
      method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({ grant_type: "fb_exchange_token", client_id: config.appId, client_secret: config.secret, fb_exchange_token: datos.access_token }),
    }));
    if (typeof extendido.access_token !== "string") throw new Error("TOKEN_INVALIDO");
    const me = new URL(`https://graph.facebook.com/${META_VERSION}/me/accounts`);
    me.searchParams.set("fields", "id,name,tasks,access_token");
    me.searchParams.set("limit", "100");
    const paginas = await respuestaMeta(await fetch(me, { headers: { Authorization: `Bearer ${extendido.access_token}` } }));
    const habilitadas = Array.isArray(paginas.data) ? paginas.data : [];
    // Por ahora se solicita seleccionar exactamente una pagina desde Meta.
    if (habilitadas.length !== 1) return pagina("Seleccioná una sola página", "<p>Volvé a conectar Facebook y autorizá exactamente una página en Meta.</p>", 422);
    const seleccionada = habilitadas[0];
    const tareas = Array.isArray(seleccionada.tasks) ? seleccionada.tasks.filter(t => typeof t === "string" && t.length <= 80).slice(0, 30) : [];
    if (!tareas.includes("CREATE_CONTENT") && !tareas.includes("PROFILE_PLUS_CREATE_CONTENT")) {
      return pagina("Permisos insuficientes", "<p>La página no tiene permisos para crear contenido.</p>", 403);
    }
    if (!/^\d{5,30}$/.test(String(seleccionada.id || ""))) throw new Error("PAGINA_INVALIDA");
    const tokenCifrado = cifrarToken(seleccionada.access_token, config.clave);
    await conDb(env, async db => {
      const pedido = await db.collection("pedidos_servicios").findOne({ pedidoId: intento.pedidoId, tipo: "publicaciones" });
      if (!pedidoHabilitado(pedido)) throw new Error("PEDIDO_NO_HABILITADO");
      const fecha = new Date();
      await db.collection(COLECCION).updateOne({ _id: `facebook:${intento.pedidoId}` }, {
        $set: {
          pedidoId: intento.pedidoId, plataforma: "facebook", estado: "conectada",
          pageId: String(seleccionada.id), nombre: String(seleccionada.name || "Página de Facebook").slice(0, 160),
          tareas, tokenPaginaCifrado: tokenCifrado, actualizadoEn: fecha,
        },
        $setOnInsert: { conectadoEn: fecha },
      }, { upsert: true });
    });
    return pagina("Facebook conectado al pedido", `<p>La página <strong>${escapar(seleccionada.name || "Facebook")}</strong> quedó vinculada exclusivamente a este pedido.</p><p>Redes en Piloto podrá programar y publicar de forma automática en tu cuenta, los contenidos que hayas brindado.</p><p>ID de página: <code>${escapar(seleccionada.id)}</code></p>`);
  } catch {
    // Nunca registrar ni devolver tokens, códigos OAuth, secretos ni respuestas completas de Meta.
    return pagina("No se pudo completar la conexión", "<p>Revisá los permisos, la página seleccionada y el estado de tu pedido.</p>", 502);
  }
}

export async function ejecutarConexionMetaPedido(request, env) {
  const url = new URL(request.url);
  if (!ORIGENES.has(url.origin)) return json({ ok: false, error: "Dominio no permitido." }, 403);
  if (url.pathname.startsWith(`${BASE}/instagram/`)) return ejecutarConexionInstagramPedido(request, env);
  if (url.pathname === CALLBACK) return callback(request, env, url);
  if (url.pathname === `${BASE}/iniciar`) return iniciar(request, env, url);
  if (url.pathname === `${BASE}/estado`) return estadoODesconectar(request, env, url, false);
  if (url.pathname === `${BASE}/desconectar`) return estadoODesconectar(request, env, url, true);
  return json({ ok: false, error: "Ruta no disponible." }, 404);
}

function responder(datos, status = 200, cookies) {
  const headers = new Headers({
    "Content-Type": "application/json; charset=utf-8",
    "Cache-Control": "no-store, max-age=0",
    "Pragma": "no-cache",
    "X-Content-Type-Options": "nosniff",
    "Referrer-Policy": "no-referrer",
  });
  if (cookies) headers.set("Set-Cookie", cookies);
  return new Response(JSON.stringify(datos), { status, headers });

const hash = (valor) => crypto.createHash("sha256").update(valor).digest("hex");
const hmacCodigo = (llave, emailHash, codigo) => crypto.createHmac("sha256", llave)
  .update(`comunidad:otp:v1:${emailHash}:${codigo}`).digest("hex");

function normalizarCorreo(valor) {
  if (typeof valor !== "string") return null;
  const correo = valor.trim().toLowerCase();
  if (correo.length < 5 || correo.length > 160 || /[\r\n\t\s]/.test(correo) ||
      !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(correo)) return null;
  return correo;
}

}

function mismaProcedencia(request) {

async function jsonPequeno(request) {
  if (!/^application\/json(?:;|$)/i.test(request.headers.get("content-type") || "")) return null;
  if (Number(request.headers.get("content-length") || 0) > 4096) return null;
  if (!request.body) return null;
  const lector = request.body.getReader();
  const decoder = new TextDecoder();
  let texto = "";
  let total = 0;
  while (true) {
    const { done, value } = await lector.read();
    if (done) break;
    total += value.byteLength;
    if (total > 4096) { await lector.cancel(); return null; }
    texto += decoder.decode(value, { stream: true });
  }
  texto += decoder.decode();
  try {
    const datos = JSON.parse(texto);
    return datos && typeof datos === "object" && !Array.isArray(datos) ? datos : null;
  } catch { return null; }
}

    await mongo.connect();
    return await callback(mongo.db(DB));
  } finally {
    await mongo.close().catch(() => {});
  }
}

// Perfil asociado al correo verificado. Nunca se asigna a una cuenta hasta
// consumir correctamente el OTP enviado a ese correo.
const PAISES_COMUNIDAD = new Set([
  "Argentina", "Belice", "Bolivia", "Brasil", "Chile", "Colombia", "Costa Rica",
  "Cuba", "Ecuador", "El Salvador", "España", "Estados Unidos", "Guatemala",
  "Haití", "Honduras", "México", "Nicaragua", "Panamá", "Paraguay", "Perú",
  "Puerto Rico", "República Dominicana", "Uruguay", "Venezuela",
]);
function validarPerfil(datos) {
  if (!datos || typeof datos !== "object" || Array.isArray(datos)) return null;
  const nombre = typeof datos.nombre === "string" ? datos.nombre.trim() : "";
  const emprendimiento = typeof datos.emprendimiento === "string" ? datos.emprendimiento.trim() : "";
  const whatsapp = typeof datos.whatsapp === "string" ? datos.whatsapp.trim() : "";
  const pais = typeof datos.pais === "string" ? datos.pais.trim() : "";
  const web = typeof datos.web === "string" ? datos.web.trim() : "";
  if (nombre.length < 2 || nombre.length > 100 || /[\r\n<>]/.test(nombre) ||
      emprendimiento.length < 2 || emprendimiento.length > 120 || /[\r\n<>]/.test(emprendimiento) ||
      !/^\+?[0-9]{8,17}$/.test(whatsapp) || !PAISES_COMUNIDAD.has(pais) ||
      web.length > 300 || /[\r\n<>]/.test(web)) return null;
  if (web) {
    try {
      const destino = new URL(/^[a-z][a-z\d+.-]*:/i.test(web) ? web : `https://${web}`);
      if (!["http:", "https:"].includes(destino.protocol) || !destino.hostname ||
          destino.username || destino.password) return null;
    } catch { return null; }
  }
  return { nombre, emprendimiento, whatsapp, pais, web };
}

async function verificarTurnstileRegistro(request, env, token) {
  const secret = String(env?.SERVICIO_TURNSTILE_SECRET ?? "").trim();
  if (!secret || typeof token !== "string" || token.length < 20 || token.length > 2048) return false;
  try {
    const ip = request.headers.get("CF-Connecting-IP");
    const body = new URLSearchParams({ secret, response: token });
    if (ip) body.set("remoteip", ip);
    const r = await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
      method: "POST", body, redirect: "manual",
    });
    if (!r.ok) return false;
    const data = await r.json();
    return data?.success === true &&
      String(data.hostname || "").toLowerCase() === new URL(request.url).hostname.toLowerCase();
  } catch { return false; }
}

async function consumirCupo(db, nombre, maximo, ahora) {
  const ventana = Math.floor(ahora / HORA_MS);
  const _id = `${nombre}:${ventana}`;
  const coleccion = db.collection("clientes_limites");
  // Fase 1 CPU: sumar y leer el contador de forma atomica en una sola consulta.
  // Los limites, su ventana y la fecha de limpieza permanecen sin cambios.
  const resultado = await coleccion.findOneAndUpdate({ _id }, {
    $inc: { solicitudes: 1 },
    $setOnInsert: { eliminarEn: new Date((ventana + 2) * HORA_MS) },
  }, {
    upsert: true,
    returnDocument: "after",
    includeResultMetadata: false,
    projection: { solicitudes: 1 },
  });
  // Compatible tanto con drivers que devuelven el documento directamente
  // como con versiones antiguas que usan el contenedor { value }.
  const actual = resultado?.value ?? resultado;
  return (actual?.solicitudes ?? maximo + 1) <= maximo;
}

async function enviarCorreo(correo, codigo, config) {
  const mensaje = `Tu codigo de acceso a Comunidad Semanal es ${codigo}. Vence en 10 minutos. Si no lo pediste, podes ignorar este mensaje.`;
  const html = `<div style="font-family:Arial,sans-serif;max-width:480px;margin:auto;color:#252525"><h2>Acceso a Comunidad Semanal</h2><p>Tu código de verificación es:</p><p style="font-size:30px;font-weight:600;letter-spacing:6px">${codigo}</p><p>Vence en 10 minutos y solo puede usarse una vez.</p><p>Si no solicitaste este código, ignorá este mensaje.</p></div>`;
  const respuesta = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: { "Authorization": `Bearer ${config.resend}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      from: config.desde,
      to: [correo],
      subject: "Código de acceso a Comunidad Semanal",
      text: mensaje,
      html,
    }),
  });
  return respuesta.ok;
}

async function solicitar(request, env, config, ipHash, dbExistente = null) {
  const datos = await jsonPequeno(request);
  const email = normalizarCorreo(datos?.email);
  if (!email) return responder({ ok: false, error: "Ingresá un correo válido." }, 400);
  const modo = datos?.modo === "registro" ? "registro" : "ingreso";
  if (modo === "registro") {
    if (!validarPerfil(datos?.perfil)) return responder({ ok: false, error: "Completá correctamente tus datos de registro." }, 400);
    if (!await verificarTurnstileRegistro(request, env, datos?.turnstileToken)) {
      return responder({ ok: false, error: "Completá nuevamente la verificación de seguridad." }, 403);
    }
  }
  const emailHash = hash(email);
  const ahora = Date.now();
  try {
    return await conDB(env, async (db) => {
      // Un pedido durante la espera de 60 segundos no debe consumir otro cupo horario.
      const codigos = db.collection("clientes_codigos");
      const anterior = await codigos.findOne({ _id: emailHash });
      if (anterior?.ultimoEnvio && ahora - new Date(anterior.ultimoEnvio).getTime() < ESPERA_ENVIO_MS) {
        return responder({ ok: false, error: "Esperá un minuto antes de pedir un nuevo código." }, 429);
      }

      // Se mantienen los cupos de 12 por IP/hora y 4 por correo/hora.
      // Los índices TTL ya fueron creados; no repetir createIndex en cada OTP.
      if (!await consumirCupo(db, `solicitar:ip:${ipHash}`, 12, ahora) ||
          !await consumirCupo(db, `solicitar:correo:${emailHash}`, 4, ahora)) {
        return responder({ ok: false, error: "Esperá antes de solicitar otro código." }, 429);
      }
      const codigo = String(crypto.randomInt(0, 1_000_000)).padStart(6, "0");
      const codigoHash = hmacCodigo(config.otpSecret, emailHash, codigo);
      const documento = {
        codigoHash, ultimoEnvio: new Date(ahora), venceEn: new Date(ahora + VIDA_CODIGO_MS),
        eliminarEn: new Date(ahora + 24 * HORA_MS), intentos: 0,
      };
      try {
        if (!anterior) {
          await codigos.insertOne({ _id: emailHash, ...documento });
        } else {
          const resultado = await codigos.updateOne({ _id: emailHash, ultimoEnvio: anterior.ultimoEnvio }, { $set: documento });
          if (resultado.modifiedCount !== 1) return responder({ ok: false, error: "Intentá nuevamente en un minuto." }, 429);
        }
      } catch (error) {
        if (error?.code === 11000) return responder({ ok: false, error: "Intentá nuevamente en un minuto." }, 429);
        throw error;
      }

      let enviado = false;
      try { enviado = await enviarCorreo(email, codigo, config); }
      catch { enviado = false; }
      if (!enviado) {
        // Conservamos los cupos consumidos para impedir abuso contra el proveedor.
        await codigos.deleteOne({ _id: emailHash, codigoHash });
        return responder({ ok: false, error: "No se pudo enviar el correo. Intentá más tarde." }, 502);
      }
      return responder({ ok: true, mensaje: "Si el correo puede recibir mensajes, llegará un código de seis dígitos." });
    }, dbExistente);
  } catch {
    return responder({ ok: false, error: "No se pudo iniciar el acceso en este momento." }, 503);
  }
}

async function verificar(request, env, config, ipHash, dbExistente = null) {
  const datos = await jsonPequeno(request);
  const email = normalizarCorreo(datos?.email);
  const codigo = typeof datos?.codigo === "string" ? datos.codigo : "";
  if (!email || !/^\d{6}$/.test(codigo)) return responder({ ok: false, error: "Revisá el correo y los seis dígitos." }, 400);
  const modo = datos?.modo === "registro" ? "registro" : "ingreso";
  const perfil = modo === "registro" ? validarPerfil(datos?.perfil) : null;
  if (modo === "registro" && !perfil) return responder({ ok: false, error: "Los datos de registro no son válidos." }, 400);
  const emailHash = hash(email);
  const ahora = Date.now();
  try {
    return await conDB(env, async (db) => {
      if (!await consumirCupo(db, `verificar:ip:${ipHash}`, 50, ahora) ||
          !await consumirCupo(db, `verificar:correo:${emailHash}`, 20, ahora)) {
        return responder({ ok: false, error: "Demasiados intentos. Solicitá un código más tarde." }, 429);
      }
      const codigos = db.collection("clientes_codigos");
      const pendiente = await codigos.findOne({ _id: emailHash });
      const fallo = () => responder({ ok: false, error: "Código incorrecto o vencido. Pedí uno nuevo si es necesario." }, 403);
      if (!pendiente?.codigoHash || pendiente.intentos >= 5 ||
          new Date(pendiente.venceEn).getTime() <= ahora) return fallo();
      const esperado = Buffer.from(pendiente.codigoHash, "hex");
      const recibido = Buffer.from(hmacCodigo(config.otpSecret, emailHash, codigo), "hex");
      if (esperado.length !== recibido.length || !crypto.timingSafeEqual(esperado, recibido)) {
        await codigos.updateOne({ _id: emailHash, codigoHash: pendiente.codigoHash, intentos: { $lt: 5 } },
          { $inc: { intentos: 1 } });
        return fallo();
      }
      // Consume el codigo de forma atomica: una peticion simultanea no puede usarlo dos veces.
      const consumido = await codigos.deleteOne({
        _id: emailHash, codigoHash: pendiente.codigoHash,
        intentos: { $lt: 5 }, venceEn: { $gt: new Date() },
      });
      if (consumido.deletedCount !== 1) return fallo();

      const cuentas = db.collection("clientes_usuarios");
      const _id = `correo:${emailHash}`;
      if (modo === "ingreso" && !await cuentas.findOne({ _id }, { projection: { _id: 1 } })) {
        return responder({ ok: false, error: "No hay una cuenta registrada con ese correo. Elegí Registrarme." }, 404);
      }
      await cuentas.updateOne({ _id }, {
        $setOnInsert: { email, creadoEn: new Date(ahora), origen: "comunidad_semanal" },
        $set: { ultimoAccesoEn: new Date(ahora), emailVerificado: true,
          ...(perfil ? { perfil, perfilActualizadoEn: new Date(ahora) } : {}) },
      }, { upsert: modo === "registro" });

      const token = crypto.randomBytes(32).toString("hex");
      const tokenHash = crypto.createHmac("sha256", config.sessionSecret).update(token).digest("hex");
      await db.collection("clientes_sesiones").insertOne({
        _id: tokenHash, usuarioId: _id,
        creadoEn: new Date(ahora), eliminarEn: new Date(ahora + VIDA_SESION_S * 1000),
      });
      const cookie = `${COOKIE}=${token}; Path=/; Secure; HttpOnly; SameSite=Lax; Max-Age=${VIDA_SESION_S}`;
      return responder({ ok: true, registrado: true, correo: email }, 200, cookie);
    }, dbExistente);
  } catch (error) {
    // Solo diagnóstico técnico: nunca registrar códigos OTP, correos ni secretos.
    console.warn("Acceso clientes: verificación fallida", {
      tipo: String(error?.name || "Error").slice(0, 80),
      codigo: Number.isInteger(error?.code) ? error.code : null,
    });
    return responder({ ok: false, error: "No se pudo completar el acceso. Intentá nuevamente." }, 503);
  }
}

function tokenSesion(request) {
  const match = (request.headers.get("cookie") || "").split(";").map(x => x.trim())
    .find(x => x.startsWith(`${COOKIE}=`));
  const token = match ? match.slice(COOKIE.length + 1) : "";
  return /^[a-f0-9]{64}$/.test(token) ? token : null;
}

async function consultarSesion(request, env, config, dbExistente = null) {
  const token = tokenSesion(request);
  if (!token) return responder({ ok: true, autenticado: false });
  try {
    return await conDB(env, async (db) => {
      const tokenHash = crypto.createHmac("sha256", config.sessionSecret).update(token).digest("hex");
      const sesion = await db.collection("clientes_sesiones").findOne({
        _id: tokenHash, eliminarEn: { $gt: new Date() },
      });
      if (!sesion) return responder({ ok: true, autenticado: false });
      const cuenta = await db.collection("clientes_usuarios").findOne({ _id: sesion.usuarioId },
        { projection: { email: 1, perfil: 1 } });
      if (!cuenta) return responder({ ok: true, autenticado: false });
      return responder({ ok: true, autenticado: true, email: cuenta.email,
        perfil: cuenta.perfil ? {
          nombre: cuenta.perfil.nombre || "", emprendimiento: cuenta.perfil.emprendimiento || "",
          whatsapp: cuenta.perfil.whatsapp || "", pais: cuenta.perfil.pais || "", web: cuenta.perfil.web || "",
        } : null });
    }, dbExistente);
  } catch { return responder({ ok: false, error: "No se pudo consultar la sesión." }, 503); }
}

async function cerrarSesion(request, env, config, dbExistente = null) {
  const token = tokenSesion(request);
  try {
    if (token) {
      await conDB(env, async (db) => {
        const _id = crypto.createHmac("sha256", config.sessionSecret).update(token).digest("hex");
        await db.collection("clientes_sesiones").deleteOne({ _id });
      }, dbExistente);
    }
    return responder({ ok: true }, 200, `${COOKIE}=; Path=/; Secure; HttpOnly; SameSite=Lax; Max-Age=0`);
  } catch { return responder({ ok: false, error: "No se pudo cerrar la sesión." }, 503); }
}

export async function ejecutarAccesoClientes(request, env, dbExistente = null, ipVerificada = null) {
  const ruta = new URL(request.url).pathname;
  const accion = ruta.startsWith(RUTA) ? ruta.slice(RUTA.length) : "";
  const permitido = { solicitar: "POST", verificar: "POST", sesion: "GET", salir: "POST" };
  if (!Object.prototype.hasOwnProperty.call(permitido, accion)) {
    return responder({ ok: false, error: "Ruta no encontrada." }, 404);
  }
  if (request.method !== permitido[accion]) return responder({ ok: false, error: "Método no permitido." }, 405);
  if (request.method === "POST" && !mismaProcedencia(request)) {
    return responder({ ok: false, error: "Origen no autorizado." }, 403);
  }
  const config = configuracion(env);
  if (!config) return responder({ ok: false, error: "Acceso por correo todavía no configurado." }, 503);
  if (accion === "sesion") return consultarSesion(request, env, config, dbExistente);
  if (accion === "salir") return cerrarSesion(request, env, config, dbExistente);
  // En el Durable Object la IP ya fue verificada por el Worker de entrada.
  // El valor adicional no se acepta directamente de Internet.
  const ip = ipVerificada || request.headers.get("CF-Connecting-IP")?.trim();
  if (!ip) return responder({ ok: false, error: "No se pudo validar el origen de la solicitud." }, 403);
  const ipHash = hash(ip);
  if (accion === "solicitar") return solicitar(request, env, config, ipHash, dbExistente);
  return verificar(request, env, config, ipHash, dbExistente);
}

// Panel de clientes registrados. No cambia los permisos de los enlaces privados
// existentes: vincular un pedido exige demostrar que se posee su clave original.
const RUTA_PANEL = "/api/servicios/clientes/panel";
const ID_PEDIDO_VALIDO = /^SER-[a-f0-9-]{36}$/i;
const CLAVE_PEDIDO_VALIDA = /^[a-f0-9]{64}$/i;

async function usuarioAutenticadoPanel(db, request, config) {
  const token = tokenSesion(request);
  if (!token) return null;
  const tokenHash = crypto.createHmac("sha256", config.sessionSecret).update(token).digest("hex");
  const sesion = await db.collection("clientes_sesiones").findOne({
    _id: tokenHash, eliminarEn: { $gt: new Date() },
  }, { projection: { usuarioId: 1 } });
  if (!sesion?.usuarioId) return null;
  const cuenta = await db.collection("clientes_usuarios").findOne({ _id: sesion.usuarioId }, {
    projection: { email: 1, emailVerificado: 1 },
  });
  return cuenta?.emailVerificado === true && typeof cuenta.email === "string"
    ? { id: sesion.usuarioId, email: cuenta.email } : null;
}

async function resumenPanel(db, usuario) {
  const documentos = await db.collection("pedidos_servicios")
    .find({ tipo: "publicaciones", clienteUsuarioId: usuario.id })
    .sort({ creadoEn: -1 }).limit(30)
    .project({ _id: 0, pedidoId: 1, estado: 1, estadoPago: 1, estadoContenidos: 1,
      estadoServicio: 1, redes: 1, creadoEn: 1, publicaciones: 1,
      modalidadContratacion: 1, prueba: 1, precioRenovacion: 1, moneda: 1,
      "negocio.nombre": 1 })
    .toArray();
  const ids = documentos.map(p => p.pedidoId).filter(x => ID_PEDIDO_VALIDO.test(x || ""));
  const conexiones = ids.length
    ? await db.collection("conexiones_meta_pedidos")
        .find({ pedidoId: { $in: ids }, plataforma: "facebook" })
        .project({ _id: 0, pedidoId: 1, nombre: 1, pageId: 1, estado: 1 }).toArray()
    : [];
  const conexionesPorPedido = new Map(conexiones.map(c => [c.pedidoId, c]));
  const pedidos = documentos.map(p => {
    const conexion = conexionesPorPedido.get(p.pedidoId);
    return {
      pedidoId: p.pedidoId,
      negocio: typeof p.negocio?.nombre === "string" ? p.negocio.nombre : "Mi emprendimiento",
      creadoEn: p.creadoEn,
      redes: Array.isArray(p.redes) ? p.redes.filter(x => x === "facebook" || x === "instagram") : [],
      estado: p.estado || "sin_estado",
      estadoPago: p.estadoPago || "pendiente",
      estadoContenidos: p.estadoContenidos || "pendiente_carga",
      estadoServicio: p.estadoServicio || "sin_activar",
      modalidad: p.modalidadContratacion || "pago",
      prueba: p.modalidadContratacion === "prueba_gratuita" ? {
        estado: p.prueba?.estado || "pendiente_activacion", inicio: p.prueba?.inicio || null,
        fin: p.prueba?.fin || null,
      } : null,
      contenidos: Array.isArray(p.publicaciones) ? p.publicaciones.map(c => ({
        id: c.id, tipo: c.tipo, estado: c.estado || "pendiente_preparacion",
      })) : [],
      facebook: conexion ? {
        nombre: conexion.nombre || "Página de Facebook",
        pageId: conexion.pageId || "",
        estado: conexion.estado || "desconocido",
      } : null,
    };
  });
  return responder({ ok: true, autenticado: true, email: usuario.email, pedidos,
    pruebaGratuitaActiva: false });
}

async function vincularPedidoPanel(db, request, usuario) {
  const datos = await jsonPequeno(request);
  const pedidoId = typeof datos?.pedidoId === "string" ? datos.pedidoId.trim() : "";
  const clave = typeof datos?.clave === "string" ? datos.clave.trim() : "";
  if (!ID_PEDIDO_VALIDO.test(pedidoId) || !CLAVE_PEDIDO_VALIDA.test(clave)) {
    return responder({ ok: false, error: "Revisá el código y la clave privada del pedido." }, 400);
  }
  const coleccion = db.collection("pedidos_servicios");
  const pedido = await coleccion.findOne({ pedidoId, tipo: "publicaciones" }, {
    projection: { _id: 1, clienteUsuarioId: 1, tokenAccesoHash: 1 },
  });
  const hashGuardado = pedido?.tokenAccesoHash;
  const calculado = hash(clave);
  if (!hashGuardado || !CLAVE_PEDIDO_VALIDA.test(hashGuardado) ||
      !crypto.timingSafeEqual(Buffer.from(calculado, "hex"), Buffer.from(hashGuardado, "hex"))) {
    return responder({ ok: false, error: "No se encontró el pedido o la clave es incorrecta." }, 404);
  }
  if (pedido.clienteUsuarioId && pedido.clienteUsuarioId !== usuario.id) {
    return responder({ ok: false, error: "Este pedido ya está vinculado a otra cuenta." }, 409);
  }
  const resultado = await coleccion.updateOne({
    _id: pedido._id, tokenAccesoHash: hashGuardado,
    $or: [{ clienteUsuarioId: { $exists: false } }, { clienteUsuarioId: usuario.id }],
  }, {
    $set: { clienteUsuarioId: usuario.id, clienteVinculadoEn: new Date() },
  });
  if (resultado.matchedCount !== 1) {
    return responder({ ok: false, error: "El pedido cambió de estado. Intentá nuevamente." }, 409);
  }
  return responder({ ok: true, vinculado: true, pedidoId });
}

// Sólo el titular de una cuenta verificada puede rotar la clave de uno de
// sus pedidos vinculados. El correo del negocio NO acredita titularidad.
async function recuperarAccesoPrivadoPanel(db, request, usuario) {
  const datos = await jsonPequeno(request);
  const pedidoId = typeof datos?.pedidoId === "string" ? datos.pedidoId.trim() : "";
  if (!ID_PEDIDO_VALIDO.test(pedidoId)) {
    return responder({ ok: false, error: "Código de pedido inválido." }, 400);
  }
  const pedidos = db.collection("pedidos_servicios");
  const pedido = await pedidos.findOne({
    pedidoId, tipo: "publicaciones", clienteUsuarioId: usuario.id,
  }, { projection: { _id: 1, tokenAccesoHash: 1 } });
  if (!pedido || !CLAVE_PEDIDO_VALIDA.test(pedido.tokenAccesoHash || "")) {
    return responder({ ok: false, error: "Pedido no encontrado o no vinculado a tu cuenta." }, 404);
  }
  // Limita las rotaciones incluso si se secuestra una sesión aún vigente.
  const permitido = await consumirCupo(db, `recuperar:usuario:${usuario.id}`, 3, Date.now());
  if (!permitido) {
    return responder({ ok: false, error: "Alcanzaste el límite de recuperaciones. Intentá más tarde." }, 429);
  }
  const tokenAcceso = crypto.randomBytes(32).toString("hex");
  const tokenAccesoHash = hash(tokenAcceso);
  const fecha = new Date();
  // La sustitución atómica invalida inmediatamente todas las claves antiguas.
  const resultado = await pedidos.updateOne({
    _id: pedido._id, clienteUsuarioId: usuario.id,
    tokenAccesoHash: pedido.tokenAccesoHash,
  }, { $set: { tokenAccesoHash, accesoPrivadoRenovadoEn: fecha, actualizadoEn: fecha } });
  if (resultado.modifiedCount !== 1) {
    return responder({ ok: false, error: "El acceso cambió recientemente. Actualizá el panel." }, 409);
  }
  // La clave solamente se devuelve en esta respuesta privada; Mongo guarda el hash.
  return responder({ ok: true, pedidoId, tokenAcceso, claveAnteriorInvalidada: true });
}

export async function ejecutarPanelClientes(request, env) {
  const ruta = new URL(request.url).pathname;
  const esResumen = ruta === RUTA_PANEL;
  const esVincular = ruta === `${RUTA_PANEL}/vincular`;
  const esRecuperar = ruta === `${RUTA_PANEL}/recuperar-acceso`;
  if (!esResumen && !esVincular && !esRecuperar) return responder({ ok: false, error: "Ruta no encontrada." }, 404);
  if (request.method !== (esResumen ? "GET" : "POST")) {
    return responder({ ok: false, error: "Método no permitido." }, 405);
  }
  if (esVincular || esRecuperar) {
    const url = new URL(request.url);
    const origin = request.headers.get("origin");
    const fetchSite = request.headers.get("sec-fetch-site");
    if (request.headers.get("x-comunidad-panel") !== "1" ||
        (origin ? origin !== url.origin : fetchSite !== "same-origin")) {
      return responder({ ok: false, error: "Origen no autorizado." }, 403);
    }
  }
  const config = configuracion(env);
  if (!config) return responder({ ok: false, error: "Panel de clientes no configurado." }, 503);
  try {
    return await conDB(env, async db => {
      const usuario = await usuarioAutenticadoPanel(db, request, config);
      if (!usuario) return responder({ ok: false, error: "Iniciá sesión desde Comunidad Semanal." }, 401);
      if (esResumen) return resumenPanel(db, usuario);
      if (esRecuperar) return recuperarAccesoPrivadoPanel(db, request, usuario);
      return vincularPedidoPanel(db, request, usuario);
    });
  } catch {
    return responder({ ok: false, error: "No fue posible consultar el panel." }, 503);
  }
}


// Solo identidad autenticada por OTP. Los invitados no se vinculan a cuentas
// por coincidencia con un correo no verificado.
// FASE7_UNA_CONEXION_DB_20261024: las rutas que ya abrieron MongoDB pueden
// autenticar dentro de esa conexion; no se reutilizan clientes entre requests.
export async function obtenerClienteRegistrado(request, env, dbExistente = null) {
  const config = configuracion(env);
  if (!config || !tokenSesion(request)) return null;
  return dbExistente
    ? usuarioAutenticadoPanel(dbExistente, request, config)
    : conDB(env, db => usuarioAutenticadoPanel(db, request, config));
}
